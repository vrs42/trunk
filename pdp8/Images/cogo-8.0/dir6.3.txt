	SUBROUTINE DIR6(KOM)
	INTEGER BUF
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	KING=KOM-62
	GO TO (299,299,15,301,303,303,303,306,303,308,309,310),(KOM-98)
299	CALL CONVRT(N1,0)
	IF(MOD(N1,2).EQ.1)GO TO 111
	IF(MOD(KING,2).EQ.0)CALL ERROR(6)
111	NMDT= NMDT+N1
	IF (KOM.EQ.99) NMDT= NMDT+N1
	CALL CONVRT(DY,1)
	CALL CONVRT(YN,1)
	IF (KOM.EQ.99) CALL CONVRT(YD,1)
	DO 300 INDEX=5,NMDT,2
	CALL CONVRT(YM,1)
	CALL CONVRT(YS,1)
300	CALL INTGRL(KING)
	GO TO 200
301	CALL CONVRT(N1,-1)
	CALL CONVRT(J,-1)
	CALL GETANG(1)
11	CALL ANGRAD
	C= A
	CALL GETANG(1)
12	CALL ANGRAD
	CALL CONVRT(K,0)
	CALL GETANG(1)
13	CALL CONVRT(M,0)
	CALL CONVRT(E1,1)
	CALL TRIANG
	GO TO 200
303	CALL CONVRT(N1,0)
	CALL CONVRT(N,-1)
	IF (KOM.EQ.107) CALL CONVRT(R,1)
	CALL CONVRT(J,-1)
14	II= KOM-102
	GO TO (302,304,305,307,307), II
302	CALL CONVRT(K,-1)
	GO TO 307
304	CALL GETANG(1)
	GO TO 307
305	CALL GETANG(3)
307	CALL OFFSET(II)
	GO TO 200
306	CALL CONVRT(N2,-1)
	CALL CONVRT(N3,-1)
	CALL CONVRT(J,-1)
	CALL CONVRT(N,-1)
	CALL CONVRT(N4,-1)
	CALL CONVRT(N1,0)
	GO TO 14
308	CALL CONVRT(N1,0)
	CALL CONVRT(NB,-1)
	CALL CONVRT(N4,-1)
	CALL CONVRT(E1,1)
	CALL CONVRT(N2,-1)
	CALL CONVRT(N5,-1)
	CALL CONVRT(E2,1)
	CALL CONVRT(E3,1)
	CALL CONVRT(R,1)
	CALL CONVRT(SIGGN,1)
	CALL DEFSPI(1)
	GO TO 200
309	CALL CONVRT(NP,0)
	CALL CONVRT(J,-1)
	CALL CONVRT(L,0)
	CALL SPICO(2)
	GO TO 200
310	CALL CONVRT(NP,0)
	CALL CONVRT(S,1)
	CALL CONVRT(L,0)
	CALL SPICO(1)
200	CALL READ
C	NEVER GETS HERE-RETURN IS A DUMMY STMNT
15	RETURN
		END
