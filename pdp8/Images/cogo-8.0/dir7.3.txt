	SUBROUTINE DIR7(KOM)
	INTEGER BUF
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	KING=KOM-62
	IF(KOM.EQ.83)GO TO 283
	IF(KOM.EQ.92)GO TO 292
	IF(KOM.EQ.93)GO TO 293
	IF(KOM.EQ.94)GO TO 294
	IF((KOM.GE.95).AND.(KOM.LT.99))GO TO 295
	IF(KOM.EQ.101)GO TO 15
	283	CALL CONVRT(N,-1)
		IF (BUF(3).NE.0) GO TO 274
		BUF(3)= 1
		E3= 0.
		N1= -N
		J= N
		GO TO 333
	274	CALL INVBR
333	NEWVAR=1
	CALL INTA
	292	BUF(3)= -1
		GO TO 278
	293	BUF(3)= -2
	278	N6= -KOM
		KOM= 41
		GO TO 199
	294	CALL CONVRT(N1,0)
		CALL CONVRT(N3,-1)
		CALL CONVRT(XN,1)
		CALL CONVRT(N4,-1)
		CALL GETANG(3)
	285	CALL CONVRT(N2,-1)
		CALL ARC(4)
C WHEN KOM IS REDEFINED,ORIG'L VALUE HELD AS NEG. INN6 (RESTORE IN INT)
	295	N6= -KOM
		IF (KING.EQ.36) GO TO 294
		KOM= KING-1
		GO TO 199
	15	N6= -KOM
		KOM= 30
199	RETURN
		END
