	SUBROUTINE CORCON
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	      DATA FILL/'_? @@A'/,X,Y/-2,-1/
C     CONVERERTS  COORDINATES FROM ONE GRID SYSTEM TO
C     ANOTHER, GIVEN A COMMON POINT AND ANGLE OF ROTATION
C
C     CONVERT ANGLE TO RADIANS
		IF (I.LE.0) CALL ERROR(16)
	      CALL ANGRAD
	      WRITE(IP,90)
	   90 FORMAT(1H ,  35HINITIAL EQUIVALENT COORDINATES ARE-)
	XX(1)=XY(J,X)
	YY(1)=XY(J,Y)
	      WRITE(IP,91) YY(1),YN,XX(1),XN
	91    FORMAT(' N-OLD=',F10.3,', N-NEW=',F10.3/
     1' E-OLD=',F10.3,', E-NEW=',F10.3/)
		CALL CKLIN(4,2)
	      CALL RADDEG
	      JMA=IABS(JM)
	      BA=ABS(B)
	      WRITE(IP,92) JD,JMA,BA
	92    FORMAT(' ANGLE OF ROTATION=',2I3,F7.3/)
		CALL CKLIN(2,2)
	      WRITE(IP,93)
	93    FORMAT('  PT',5X,'N-OLD',5X,'E-OLD',5X,'N-NEW',5X,'E-NEW')
	      XFAC = XN-XY(J,X)*COS(A)+XY(J,Y)*SIN(A)
	      YFAC = YN-XY(J,Y)*COS(A)-XY(J,X)*SIN(A)
	      DO 1 IN = I,K
	      IF((XY(IN,X).EQ.FILL).OR.(XY(IN,Y).EQ.FILL)) GO TO 1
	      XT = XY(IN,X)
	      YT = XY(IN,Y)
	      YY(1) = YFAC+YT   *COS(A)+XT   *SIN(A)
	      XX(1) = XFAC+XT   *COS(A)-YT   *SIN(A)
	CALL XYOUT(IN,X,XX(1))
	CALL XYOUT(IN,Y,YY(1))
	      WRITE(IP,100) IN,YT,XT,YY(1),XX(1)
	100   FORMAT(I4,4F10.3)
	      CALL CKLIN(1,2)
	    1 CONTINUE
		CALL CKLIN(1,2)
	      CALL READ
	      STOP
	      END
