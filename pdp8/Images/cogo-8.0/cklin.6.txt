	SUBROUTINE CKLIN(NO,ISWX)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	COMMON/QQQ/LN1,LN2,NPG1,NPG2
		DIMENSION HOLLY(9)
		EQUIVALENCE (HOLL(1),HOLLY(1))
C ** DESCRIPTION OF ARGUMENTS
C
C	NO	= -1 ,GO TO BOTTOM OF PAGE,
C		=  0,PRINT NEW PAGE HEADING
C		= + ,DECREMENT LINE COUNTER BY 'NO' & TEST FOR PAGE END
C
C	ISWX	= 1,OUTPUT TO TTY ONLY,
C		= 2, OUTPUT TO FILE,
C		= 3,OUTPUT BOTH TO TTY & TO FILE
C		= 4,INITIALIZES LINE & PAGE COUNTERS ONLY
C	NOTE:	IF OUTPUT NOMINALLY TO FILE IS DIRECTED INSTEAD TO
C		THE TTY, THIS ROUTINE WILL MAKE THE APPROPRIATE PAGING
C
	ISW= ISWX
		IF ((IP.EQ.4).AND.(ISW.EQ.2.OR.ISW.EQ.3)) ISW= 1
		GO TO (100,200,100,300), ISW
	100	IF (NO) 110,150,130
110	DO 115 LNN1=1,LN1
	115	WRITE(4,420)
		LN1= 1
	420	FORMAT(' ')
		GO TO 135
	300	LN1= ILIN-4
		LN2= ILIN-4
		NPG1= 0
		NPG2= 0
		GO TO 400
	130	LN1= LN1-NO
	135	IF (LN1.GT.0) GO TO 170
	150	LN1= ILIN-4
		NPG1= NPG1+1
		WRITE(4,460)HOLLY,NPG1
	460	FORMAT(1H1/,1X,9A6,8X,' PG',I3///)
		WRITE(4,420)
	170	GO TO (400,200,190),ISW
	190	IF (IP.EQ.4) GO TO 400
	200	IF (NO) 210,250,230
210	DO 215 LNN1=1,LN2
	215	WRITE(IP,420)
		LN2= 1
		GO TO 235
	230	LN2= LN2-NO
	235	IF (LN2.GT.0) GO TO 270
250		LN2= ILIN-4
		NPG2= NPG2+1
		WRITE(IP,460)HOLLY,NPG2
		WRITE(IP,420)
	270	CONTINUE
	400	RETURN
		END
