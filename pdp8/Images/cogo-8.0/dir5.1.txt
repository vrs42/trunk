	SUBROUTINE DIR5(KOM)
	INTEGER BUF
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	KING=KOM-62
	GO TO (284,284,286,287,288,289,290,291),KOM-83
284	CALL CONVRT(J,-1)
		CALL CONVRT(N,-1)
		CALL DSTRAD
		IF (KING.EQ.22) A= DX
		IF (KING.EQ.23) A= DY
		WRITE(IP,84) A
	84	FORMAT(' DIST=',F10.3)
		CALL CKLIN(1,2)
		GO TO 200
	286	CALL CONVRT(J,-1)
		CALL CONVRT(B,1)
		CALL CONVRT(N,-1)
		CALL CONVRT(A,1)
		B= A-B
		CALL DSTRAD
		A= DIST
		CALL TRI(1.)
	287	CALL CONVRT(AREA,1)
		CALL CONVRT(B,1)
		AREA= AREA+B
		CALL CONVRT(N1,0)
		CALL SAVE(16)
	288	CALL CONVRT(DIST,1)
		CALL CONVRT(B,1)
		DIST= DIST+B
		CALL CONVRT(N1,0)
		T= DIST
		CALL SAVE(14)
	275	N1= -1
		D= A
		CALL GETANG(1)
		GO TO 276
	289	CALL GETANG(3)
	276	CALL ANGRAD
		CALL NEWAZ
		IF (N1.GE.0) GO TO 275
		A= A+D
		CALL CONVRT(N1,0)
		CALL SAVE(19)
	291	CALL CONVRT(N1,0)
	290	CALL CONVRT(J,-1)
		CALL CONVRT(N,-1)
		CALL DSTRAD
		YN= DIST
		T= 0.
		XL= 0.
		JD= 0
		JM= 0
		SC= 0.
		CALL CONVRT(R,1)
		IF (R.LT.0.) N1= -N1
	CALL CURVE(KING)
200	CALL READ
	END
