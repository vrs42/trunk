	SUBROUTINE DIR1(KOM)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA MINUS1/-1/
	GO TO (201,201,204,205,205,205,205,210,209,210,211,211
     1,209,209,209,205,209,209,209,220,221,222,209,209),KOM
201	CALL CONVRT(N,0)
	CALL CONVRT(J,0)
	IF(KOM .EQ. 1) CALL CLR
	CALL CONVRT(I,0)
	CALL DMP
204	CALL CONVRT(N,0)
	CALL CONVRT(A,1)
	CALL CONVRT(B,1)
	CALL STR
205	II= MOD(KOM,4)
		CALL CONVRT(J,MINUS1)
	JJ= MOD(II,2)
	JJ= -JJ
	CALL CONVRT(N,JJ)
	IF (KOM.EQ.16) CALL REDFN
	IF (JJ.LT.0) GO TO 206
	CALL CONVRT(DIST,1)
	CALL GETANG(II+1)
206	GO TO (306,207,208,304), II+1
306	CALL LOCAZ
207	CALL INVAZ
	GO TO 200
208	CALL LOCBR
304	CALL INVBR
		GO TO 200
210	CALL CONVRT(N,MINUS1)
	CALL CONVRT(J,MINUS1)
	CALL CONVRT(I,0)
	CALL CONVRT(YN,1)
	CALL GETANG(1)
	CALL LOCATE(KOM/5)
211	CALL CONVRT(I,MINUS1)
	CALL CONVRT(J,MINUS1)
	CALL CONVRT(K,MINUS1)
	IF (KOM.EQ.12) CALL CONVRT(N1,0)
	CALL ANG(KOM-10)
209	CALL CONVRT(J,MINUS1)
	CALL CONVRT(N,MINUS1)
	IF (KOM-13) 212,213,214
212	CALL CONVRT(I,0)
213	CALL CONVRT(XN,1)
	IF (KOM.EQ.13) GO TO 214
	CALL LOCLN
	GO TO 200
214	II= KOM-13
	IF (II.EQ.1) GO TO 215
	IF (II.GE.4) GO TO 217
	CALL CONVRT(K,0)
215	IF (II.NE.0) CALL DISTAN(II)
	CALL CONVRT(L,0)
	CALL PARLN
217	ARG= 1.
	IF (II-6) 219,218,223
218	ARG= -ARG
219	CALL CONVRT(XL,1)
	CALL SEGMT(ARG,II-3)
223	CALL CONVRT(M,0)
	IF (II.EQ.11) CALL CONVRT(N1,0)
	CALL PTLOC(14-II)
220	CALL CONVRT(N1,0)
	CALL CONVRT(N4,MINUS1)
	CALL CONVRT(E1,1)
	CALL CONVRT(N2,MINUS1)
	CALL CONVRT(N5,MINUS1)
	CALL CONVRT(SPT,1)
	CALL CONVRT(N3,MINUS1)
	CALL CONVRT(SIGGN,1)
	CALL DEFCUR
221	CALL CONVRT(NP,0)
222	CALL CONVRT(N,MINUS1)
	CALL CONVRT(N1,0)
	II= KOM-20
	IF (II.GT.20) II= 47-II
	CALL PTLOC(II)
200	CALL READ
	END
