	SUBROUTINE PTLOC(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA NCURV,NPC,SPC,NPT,NC/1,2,3,4,6/
C     II=1,OFFSET/ALIGN
C     II=2 STATION/POA
C     II=3 DIVIDE/STATION/LINE
C     II=4 DIVIDE/LINE
C     II=5 OFFSET/EL/ALIGN
C     II=6 STATION/EL/POA
	      IF (II-4) 11,3,11
	   11 NN= N
	      KJ= J
	 10   DO 30 K=1,IA
	      IF(N1-XY(K,NCURV))30,40,30
	 30   CONTINUE
	      CALL ERROR(19)
	 40   CALL RECALC
	      N= NN
	      J= KJ
	      GO TO 50
	 20   K=IPRES
	 50   GO TO(1,2,3,3,1,2),II
	 1    JOE=1
	 529  LBJ=1
	      GO TO 500
	 510  GO TO (511,516,512),MCURV
C     OFFSET TO CURV
	 511  DIST=R
	      J=XY(K,NC)
	 514  GO TO (527,528,585,586),JOE
	527   WRITE(IP,404) NP,N,S,OFF
	404   FORMAT(' RAD PT=',I3,', KN PT=',I3,', STA=',F10.3,
     1', OFFSET=',F10.3)
	      CALL CKLIN(1,2)
	      N=NP
	      CALL NEWPT(1)
	      IF(II-4)400,400,401
	 401  CALL VERPT(6)
	 400  CALL READ
C     OFFSET TO BACK TANGENT
	 512  J=XY(K,NPC)
	      XN=HBA
	      YN=XY(K,SPC)
	      B=-1.0
	 515  CALL DSTRAD
	 513  C=XN-A
	      OFF=-DIST*SIN(C)*B
	      DIST=ABS(DIST*COS(C))
	      S=YN+DIST*B
	      A=XN
	      GO TO 514
C     OFFSET TO FORWARD TANGENT
	 516  J=XY(K,NPT)
	      XN=HAA
	      YN=SPT
	      B=1.0
	      GO TO 515
C     STATION/POA ROUTINE
	 2    JOE=2
	      GO TO 529
528	ZNEW=XY(K,NCURV)
	   WRITE(IP,88) ZNEW,N,S
	88    FORMAT(' CURVE NO.=',I3,', PT=',I3,', STA=',F10.3)
	      CALL CKLIN(1,2)
	      IF(II-2)400,400,401
C     ROUTINE TO COMPUTE STATION AND OFFSET OF A POINT TO CURVE
C     AND TO FIND OUT IF OFFSET POINT IS ON BACK TANGENT,
C     CURVE, OR FORWARD TANGENT
C     ENTER WITH N=IDENT OF POINT OF KNOWN COORDINATES
	 500  J=XY(K,NC)
	      CALL DSTRAD
	501   C=(A-APC)*SIGGN
	      IF(C)506,502,502
	 502  IF(C-3.1415926)503,508,508
	 503  S=XY(K,SPC)+R*C
	      IF(S-SPT)504,504,505
	 504  MCURV=1
	      OFF=(R-DIST)*SIGGN
	      GO TO 509
	 505  MCURV=2
	      GO TO 509
	 506  IF(ABS(C)-3.1415926) 508,508,507
	 507  C=C+6.2831853
	      GO TO 503
	 508  MCURV=3
	 509  GO TO 510
C     DIVIDE/LINE ROUTINE
	 3    CALL DSTRAD
	 583  XM=M
	      XM=DIST/XM
	      WRITE(IP,408) XM
	408   FORMAT(' DIST BETWEEN PTS=',F10.3)
	      CALL CKLIN(1,2)
	      SA=A
	      YM=0.0
	      KJ=J
	      KN=N
	      N=J
	      JOE=3
	      DO 584 I=1,M
	      DIST=YM
	      A=SA
	      CALL NEWPT(1)
	      IF(II-3)529,529,70
	 585  WRITE(IP,444) S,OFF
	      CALL CKLIN(1,2)
	 70   J=KJ
	      N=KJ+I
	 584  YM=YM+XM
	      N=KN
	      CALL PRINT
	      JOE=4
	      IF(II-3)529,529,80
	586   WRITE(IP,444) S,OFF
	444   FORMAT(' STA=',F10.3,', OFFSET=',F10.3)
	      CALL CKLIN(1,2)
	 80   CALL  READ
	      END
