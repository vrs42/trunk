C	IW=1 FOR INPUT,2 FOR OUTPUT.
C	FOR PPG,PPS COMMAND
C	WRITES SMALL FILE ONLY,READS BOTH.

	SUBROUTINE POINTS(IW,IQ,N,J)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	DIMENSION BUF(85)

	IF((N.LE.0).OR.(J.GT.999)) GO TO 20
	IF(N.GT.J) GO TO 20
	WRITE(2'XYPOS)ARRAYX
	IT2=IFIX((N-1)/42)
	IT3=2*(N-IT2*42)-1
	REWIND IQ
	GO TO(2,13),IW
2	READ(IQ) BUF
	IF((N.LT.BUF(1)).OR.(J.GT.BUF(2))) GO TO 20
	IT4=(N-BUF(1))*2+3
21	IF(IT4.LT.85) GO TO 1
	READ(IQ) BUF
	IT4=IT4-84
	GO TO 21
7	ARRAYX(IT3)=BUF(IT4)
	ARRAYX(IT3+1)=BUF(IT4+1)
	IT3=IT3+2
	IT4=IT4+2
	N=N+1
	IF(N.GT.J) GO TO 5
	IF(IT3.NE.85) GO TO 6
10	WRITE(2'IT2)ARRAYX
	IT3=1
1	IT2=IT2+1
	READ(2'IT2)ARRAYX
6	IF(IT4.NE.85) GO TO 7
12	READ(IQ)BUF
	IT4=1
	GO TO 7
17	WRITE(IQ) BUF
5	XYPOS=IT2
	CALL READ
13	IT4=3
	BUF(1)=N
	BUF(2)=J
	GO TO 16
15	BUF(IT4)=ARRAYX(IT3)
	BUF(IT4+1)=ARRAYX(IT3+1)
	IT4=IT4+2
	IT3=IT3+2
	N=N+1
	IF(N.GT.J) GO TO 17
	IF(IT3.NE.85) GO TO 14
	IT3=1
16	IT2=IT2+1
	READ(2'IT2) ARRAYX
14	IF(IT4.NE.85) GO TO 15
	IT4=1
	WRITE(IQ) BUF
	GO TO 15
20	CALL ERROR(15)
	END
