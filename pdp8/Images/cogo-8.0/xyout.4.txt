C	CALLING SEQUENCE FOR XYOUT AND XY.
C	V=ARRAY INDEX
C	P=-1 FOR Y
C	P=-2 FOR X
C	P; 1=NCURV,2=NPC,3=SPC,4=NPT,5=NPI,6=NC,
C	   7=SVPI,8=ZVPI,9=NVPI,10=XA,11=XB,12=YA
C	   13=YB,14=YC,15=RA,16=RB,17=RC,18=RSAVE,
C	   19=ASAVE,20=DSAVE,21=IE
C
C	CALL XYOUT(V,P,QUANT); WHERE QUANT=OUTPUT VAL
C	DUM=XY(V,P)
	SUBROUTINE XYOUT(V,P,QUANT)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
C	CALL FUNCTION TO SET UP POINTS
	DUMMY=XY(V,P)
	IF (V.GE.1000) GO TO 1
	ARRAYX(IPX)=QUANT
	RETURN
1	XY3(IPX)=QUANT
	RETURN
	END
