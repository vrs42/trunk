	SUBROUTINE TANG(CROSS)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	      SIGGN= N2
	      N= N1
	      CALL DSTRAD
	      B= XL-SC*CROSS
	      IF (DIST-B) 442,446,447
	  442 SIZE= DX
	      IF (DX-DY) 443,444,444
	  443 SIZE= DY
	  444 CHECK= SIZE*1.5E-07
	      IF (DIST-B+CHECK) 445,445,447
	446	IF(B) 445,445,447
	  445 WRITE(4,10)
   10 FORMAT(' PT OF TAN OF ONE CIRCLE FALLS WITHIN OTHER CIRCLE')
	      CALL CKLIN(1,1)
	  447 DIST= SQRT(DIST**2-B**2)
	      ANG= ATAN(B/DIST)
	      E1= A+ANG*SIGGN
	      A= E1-1.5707963*SIGGN
	      N= M
	      DIST= XL
	      CALL NEWPT(1)
	      J= N1
	      N= K
	      DIST= SC*CROSS
	      CALL NEWPT(1)
	      IF (E1) 457,456,454
	  457 E1= E1+6.2831853
	      GO TO 456
	  454 IF (E1-6.2831853) 456,455,455
	  455 E1= E1-6.2831853
	      GO TO 454
	  456 J= M
	      N= K
	      CALL DSTRAD
	      CALL RADDEG
	      WRITE(IP,20) J,N,DIST,JD,JM,B
	   20 FORMAT(' PT',I4,' TO PT',I4,', DIST=',F10.3,', AZ='
     1,2I3,F7.3)
	      CALL CKLIN(1,2)
	      CALL READ
	      END
