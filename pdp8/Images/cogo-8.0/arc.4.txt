	SUBROUTINE ARC(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
		IF (N1.LE.0) CALL ERROR(16)
C     ARC/ARC/INTERSECT
	      GO TO(1,2,3,3),II
1	DX=XY(N3,X)-XY(N4,X)
	DY=XY(N3,Y)-XY(N4,Y)
	      IDY=DY
		DIST= SQRT(DX*DX+DY*DY)
	IF ((DIST.GT.(XN+YN)).OR.(DIST.LT.ABS(XN-YN))) CALL ERROR(28)
	      IF(IDY.NE.0) GO TO 876
	 875  T5=-(DX*DX+XN*XN-YN*YN)/2./DX/XN
	      T6=T5
	      T9=SQRT(1.0-T5**2)
	      T10=-T9
	      GO TO 880
	 876  T1=2.0*XN*DX
	      T2=2.0*XN*DY
	      T3=XN**2-YN**2+DX**2+DY**2
	 998  T4=T1**2+T2**2
	      T11=2.0*T1*T3
	      T12=T3**2-T2**2
C     SOLUTION OF QUAD. EQUAT.
	      T7=SQRT(ABS(T11**2-4.0*T4*T12))/(2.0*T4)
	      T8=-T11/(2.0*T4)
	      T11=T8+T7
	      T12=T8-T7
		T5=T11
		T6=T12
	      T9=(-T3-T1*T11)/T2
	      T10=(-T3-T1*T12)/T2
880	XY3(3)=XY(N3,X)+XN*T5
	XY3(4)=XY(N3,Y)+XN*T9
	XY3(5)=XY(N3,X)+XN*T6
	XY3(6)=XY(N3,Y)+XN*T10
C     TEST FOR CORRECT SOLUTION
	      N=1001
	      J=N2
	      CALL DSTRAD
	 123  YN=DIST
	      N=1002
	      CALL DSTRAD
	 120  IF(DIST-YN)122,121,121
	 121  N2=1001
	      GO TO 125
	 122  N2=1002
125	CALL XYOUT(N1,X,XY3((N2-1000)*2+1))
	CALL XYOUT(N1,Y,XY3((N2-1000)*2+2))
	      N=N1
	      CALL PRINT
		IF (N6.GE.0) GO TO 5
		N=J
		I=J
		J=N3
		CALL LOCLN
		J=N
		N=N1
		CALL INVBR
		A= 2.*ASIN(DIST/2./XN)
		XL= A*XN
		CALL RADDEG
		WRITE(IP,10) XL,JD,JM,B
	10	FORMAT(' ARC LENGTH=',F10.3,'; DELTA=',2I3,F7.3)
		CALL CKLIN(1,2)
	5	CALL READ
C     ARC/LINE/POINTS
	 2    N=N5
	      J=N4
	      CALL DSTRAD
		GO TO 4
	6	IF (ABS(COS(A)).GE..99999990) GO TO 870
	871	IF (ABS(SIN(A)).GE..99999990) GO TO 872
	 118  SINA=SIN(A)
	      COSA=COS(A)
	      GO TO 874
870	DX=XY(N4,X)-XY(N3,X)
	      T5=DX/XN
	      T6=T5
	      T9=SQRT((XN**2)-(DX**2))/XN
	      T10=-T9
	      GO TO 880
872	DY=XY(N4,Y)-XY(N3,Y)
	      T5=SQRT((XN**2)-(DY**2))/XN
	      T6=-T5
	      T9=DY/XN
	      T10=T9
	      GO TO 880
	 874  T1=XN/SINA
	      T2=-XN/COSA
	T3=(XY(N3,X)-XY(N4,X))/SINA-(XY(N3,Y)-XY(N4,Y))/COSA
	      GO TO 998
C     ARC/LINE/AZ
	 3    CALL ANGRAD
		IF (II.EQ.4) CALL NEWAZ
		J=N4
	4	N= N3
		B= A
		CALL DSTRAD
		A= A-B
		IF (ABS(DIST*SIN(A)).GT.XN) CALL ERROR(28)
		A= B
		IF (II.EQ.2) GO TO 6
	      SINA=SIN(A)
	      COSA=COS(A)
	      IF(ABS(COSA)-.99999990)300,870,870
	 300  IF(ABS(SINA)-.99999990)874,872,872
	      END
