	SUBROUTINE STOSUP(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	COMMON/MMM/MN,DD
	DATA SVPI,YA,YB,YC,RA,RB,RC/7,12,13,14,15,16,17/
	      GO TO (1,2),II
	 1    IF(SVB-XY(1,SVPI))10,20,20
	 20   IF(SVA-XY(NTOT,SVPI))30,30,40
	 30   MN=0
	      IF (N3.GT.10) GO TO 15
	      ID=1
C     ID=1 INDICATES FUTURE COMMANDS BASED ON THIS COMMAND CAN BE
C     EXECUTED
	      DD=SVA-SVB
	      CALL READ
C     READ DIST,C,OFF,XAA
	 2    IF(ID)3,5,4
C     WAS STORE/SUPER COMMAND OK
	   15 WRITE(4,500)N3
	  500 FORMAT(I5,' SEGMENTS EXCEEDS MAX ALLOWED')
	   17 CALL CKLIN(1,1)
	      IS= 4
	      CALL READ
	 4    MN=MN+1
	      IF (MN.LE.N3) GO TO 16
	      WRITE(4,500) MN
	      GO TO 17
16	CALL XYOUT(MN,YB,DIST)
	CALL XYOUT(MN,RB,C)
	CALL XYOUT(MN,YA,OFF)
	CALL XYOUT(MN,RA,XAA)
	CALL XYOUT(MN,YC,(OFF-DIST)/DD)
	CALL XYOUT(MN,RC,(XAA-C)/DD)
	      CALL READ
	 10   ID=-1
C     SET FLAG TO SHOW SURVEY/STATION COMMAND WRONG
	      CALL ERROR(21)
	 40   ID=-1
	      CALL ERROR(22)
	 3    CALL ERROR(23)
	 5    CALL ERROR(25)
	      END
