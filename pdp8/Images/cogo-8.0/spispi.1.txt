	SUBROUTINE SPISPI
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
		IF (N3.LE.0) CALL ERROR(16)
	      N1=1000
	      N2=NB
	      CALL INTSCT(2,2)
	      SINB=SIN(YD)
	      COSB=COS(YD)
	      SINA=SIN(A)
	      COSA=COS(A)
	      T1=1.0/(2.0*E3*E1)
	      P1=1.0/(2.0*BA*E4)
	XS=XY(NB,X)
	YS=XY(NB,Y)
	XP=XY(N4,X)
	YP=XY(N4,Y)
	      N=1000
	      J=NB
	      CALL DSTRAD
	      XL=DIST
	      J=N4
	      CALL DSTRAD
	      D=DIST
		DO 89 KT=1,20
	      T2=XL-XL**5*T1**2/10.0
	      T3=XL**3*T1/3.0-XL**7*T1**3/42.0
	      T4=1.0-XL**4*T1**2/2.0
	      T5=XL**2*T1-XL**6*T1**3/6.0
	      P2=D**5*P1**2/10.0-D
	      P3=D**7*P1**3/42.0-D**3*P1/3.0
	      P4=D**4*P1**2/2.0-1.0
	      P5=D**6*P1**3/6.0-D**2*P1
	      F1=T2*SINB+T3*COSB+P2*SINA+P3*COSA+XS-XP
	      F2=T2*COSB-T3*SINB+P2*COSA-P3*SINA+YS-YP
	      PF1L=T4*SINB+T5*COSB
	      PF2L=T4*COSB-T5*SINB
	      PF1D=P4*SINA+P5*COSA
	      PF2D=P4*COSA-P5*SINA
C     COMPUTE AND MAKE CORRECTIONS
	      DELL=(PF1D*F2-PF2D*F1)/(PF1L*PF2D-PF1D*PF2L)
	      DELD=-(PF2L*DELL+F2)/PF2D
	      XL=XL+DELL
	      D=D+DELD
	      IF(ABS(DELL)+ABS(DELD)-.01)88,88,89
	89	CONTINUE
	 88   S1=T+XL
	      S2=OFF+D
	XX(1)=-P2*SINA-P3*COSA+XP
	YY(1)=-P2*COSA+P3*SINA+YP
	CALL XYOUT(N3,X,XX(1))
	CALL XYOUT(N3,Y,YY(1))
	      WRITE(IP,10) N3,YY(1),XX(1),S1,NB,XL,S2,N4,D
	10    FORMAT(' INTERSECTION PT=',I3,', N=',F10.3,', E=',F10.3/
     1' STA SPIRAL INT 1=',F10.3,', DIST (PT',I4,' TO INT)=',F10.3/
     2' STA SPIRAL INT 2=',F10.3,', DIST (PT',I4,' TO INT)=',F10.3)
	      CALL CKLIN(3,2)
	      CALL READ
	      END
