	SUBROUTINE VERPT(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA SVPI,ZVPI,NVPI,XA,XB,YA,YB,YC,RA,RB,RC/7,8,9,10,11
     1,12,13,14,15,16,17/
C     II=1,SURVEY/STATION
C     II=2,EVEN/STATIONS
C     II=3 OFFSET/ELEV
C     II=4 SUPER/SPEC
C     II=5,SUPER/EVEN
C     II=6 ELEVATIONS FOR XY COORDS
	      N= NSAVE
	      IF(XY(1,XB))1,1,2
C     CHECK TO SEE IF CURVE DEFINED
	 2    GO TO (562,11,562,562,11,562),II
C     EVEN/STATIONS READ,S,OFF,E3
	 11   IF(S-XY(1,SVPI))560,562,562
	  560 WRITE(4,100) S
	      S= S+OFF
	100   FORMAT(' STA',F10.3,' IS TOO SMALL')
	      CALL CKLIN(1,1)
	      GO TO 11
C     CHECK IF STATION IS WITHIN PRESENTLY STORED CURVE
	 562  IF(S-XY(1,SVPI))110,130,120
	 120  IF(S-XY(NTOT,SVPI))130,130,200
	 130  IF(S-SVPC)505,504,503
	 503  IF(S-SVPT)504,504,602
	 504  N=NSAVE
	      GO TO 26
	 505  IF(NSAVE-2)25,25,605
	 605  NSAVE=2
	      GO TO 502
	  602 IF (NTOT-NSAVE-1) 502,499,502
	  502 NN= NTOT-1
	      DO 565 N=NSAVE,NN
	      IF (S.LE.(XY(N,SVPI)+XY(N,XA))) GO TO 99
	  565 CONTINUE
	      GO TO 499
	  990 CALL ERROR(18)
	 99   CALL PCPT
	 24   IF(S-SVPC)25,25,26
	 25   Z=XY(N-1,ZVPI)+(S-XY(N-1,SVPI))*GB
	      G=GB*100.0
	      GO TO 20
	  499 N= NTOT-1
	      CALL PCPT
	   49 Z= XY(NTOT,ZVPI)-(XY(NTOT,SVPI)-S)*GA
	      G= GA*100.
	      GO TO 20
	 26   IF(S-XY(N,SVPI))27,27,29
	 27   Z=((S-SVPC)**2)*AB+(S-SVPC)*GB+ZVPC
	      G=(2.0*(S-SVPC)*AB+GB)*100.0
	      GO TO 20
	 29   IF(S-SVPT)291,291,292
	 292  N=N+1
	      IF (NTOT-N) 990,499,99
	 291  Z=((S-SVPT)**2)*BB+(S-SVPT)*GA+ZVPT
	      G=(2.0*(S-SVPT)*BB+GA)*100.0
	      GO TO 20
	 1    CALL ERROR(24)
	 110  CALL ERROR(21)
	 200  CALL ERROR(22)
	 20   NSAVE=N
	 21   GO TO(7,6,8,9,9,12),II
	6     WRITE(IP,10) S,Z,G
	10    FORMAT(' STA=',F10.3,', HWY EL=',F9.3,', GRADE=',F7.3)
	      CALL CKLIN(1,2)
	 22   S=S+OFF
C     HAVE WE EXCEEDED LARGEST STATION SPECIFIED IN COMMAND
	      IF(S-E3)24,24,400
	 400  CALL READ
	12    WRITE(IP,600) Z
	600   FORMAT(' HWY EL=',F9.3)
	CALL CKLIN(1,2)
	      RETURN
C     SURVEY/STATION READ S,E5
	 7    COTA=Z-E5-DIFF
	      WRITE(IP,300) S,Z,E5,COTA,G
300   FORMAT(' STA=',F10.3,', HWY EL=',F9.3,', TERRAIN EL=',F9.3/
     1' COTA=',F7.3,', GRADE=',F7.3)
	      CALL CKLIN(2,2)
	      CALL READ
C     OFFSET/ELEV READ S,OFF,E4
	 8    ZX=Z+ABS(OFF)*(E4/100.0)
	      WRITE(IP,50) S,Z,OFF,E4,ZX
50    FORMAT(' STA=',F10.3,', C/L EL=',F9.3,', OFFSET DIST=',F9.3/
     1' CROSS SLOPE=',F7.3,', OFFSET EL=',F9.3)
	      CALL CKLIN(2,2)
	      CALL READ
C     SUPER/SPECIAL READ S
	9     WRITE(IP,60) S,Z
	60    FORMAT(' STA=',F10.3,', C/L EL=',F9.3)
	      CALL CKLIN(1,2)
	      DO 211 I=1,N3
	      IF(S-SVB)80,80,81
	 80   DIST=XY(I,YB)
	      RATE=XY(I,RB)
	      GO TO 19
	 81   IF(S-SVA)18,82,82
	 82   DIST=XY(I,YA)
	      RATE=XY(I,RA)
	      GO TO 19
18	DIST=XY(I,YB)+(S-SVB)*XY(I,YC)
	      RATE=XY(I,RB)+(S-SVB)*XY(I,RC)
	 19   Z=Z+DIST*RATE/100.0
	      WRITE(IP,1000) I,DIST,RATE,Z
	1000  FORMAT(' SEG NO=',I3,', SEG WIDTH=',F9.3,', CROSS SLOPE=',
     1 F9.3,', EL=',F9.3)
	211	CALL CKLIN(1,2)
	      IF(II-5)400,22,400
C     SUPER/EVEN READ S,OFF,E3
	      END
