	SUBROUTINE CRVSPI
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
		IF ((N1.LE.0).OR.(N2.LE.0).OR.(N5.LE.0)) CALL ERROR(16)
		 ITEST=1
	N=L
		 J=NB
		 CALL DSTRAD
		 BA=A
		 J=NA
		 CALL DSTRAD
		 AA=A
		 F=BA-AA
		 IF(ABS(F)-3.1415926)34,33,33
	33	 F=F-6.2831853*F/ABS(F)
	34	 IF(F)35,36,36
	35	 E3=-E2
		 GO TO 37
	36	 E3=E2
	37	 CV=ABS(F)
		 FF=CV
		 A=CV
		 CALL RADDEG
		 WRITE(IP,10) N6,L,JD,JM,B
	10	 FORMAT(' CURVE',I4,', PI=',I3,', VERT,ANGLE AT PI= '
     1,2I3,F7.3)
		 CALL CKLIN(1,2)
	17	 TS=XL/(2.0*E3)
	18	 ATS=ABS(TS)
		 XC=XL*(1.0-ATS**2/10.0)
		 YD=XL*(ATS/3.0-ATS**3/42.0)
		 SINA=SIN(ATS)
		 COSA=COS(ATS)
		 XST=YD/SINA
		 XLT=XC-YD*COSA/SINA
		 P= YD-E2*(1.-COSA)
		 XK=XC-E2*SINA
		 CV=CV+ATS
		 GO TO (21,22),ITEST
	21	 D1=E2+P
		 CA=BA+TS
		 CB=CA
	3	 A=ATS
		 CALL RADDEG
		 WRITE(IP,15)
	15	 FORMAT(13H FIRST SPIRAL)
	32	 WRITE(IP,20)  XL,JD,JM,B,XST,XLT,XC,YD,P,XK
	20	 FORMAT(' LENGTH=',F10.3,',TOT SPIRAL ANGLE=',2I3,F7.3,
     1',SHORT TAN=',F10.3/' LONG TAN=',F10.3,',SPIRAL PROJ ON TAN='
     2,F10.3/' END SPIRAL TO TAN=',F10.3,',P=',F10.3,',K=',F10.3)
		 CALL CKLIN(4,2)
		 GO TO (12,900,23),ITEST
	12	 SAVE=XL
		 XL=OFF
		 TS=-OFF/(2.0*E3)
		 XK1=XK
		 EE1= XLT
		 EE2= XST
		 ITEST=2
		 GO TO 18
	22	 D2=E2+P
		 E=ATAN(D2*SIN(FF)/(D1+D2*COS(FF)))
		 TTS=D1*COS(FF-E)/SIN(FF-E)+XK1
		 TTS1=D2*COS(E)/SIN(E)+XK
23	ZNEW=XY(L,X)-TTS*SIN(BA)
	CALL XYOUT(N1,X,ZNEW)
		 ZNEW=XY(L,Y)-TTS*COS(BA)
	CALL XYOUT(N1,Y,ZNEW)
		 ZNEW=XY(N1,X)+EE1*SIN(BA)+EE2*SIN(CA)
	CALL XYOUT(N2,X,ZNEW)
		 ZNEW=XY(N1,Y)+EE1*COS(BA)+EE2*COS(CA)
	CALL XYOUT(N2,Y,ZNEW)
	31	 WRITE(IP,30) TTS
	30	 FORMAT(24H TOTAL TANGENT DISTANCE=F10.3)
		 CALL CKLIN(1,2)
		 GO TO (900,41,38),ITEST
	41	 ITEST=3
		 WRITE(IP,40)
	40	 FORMAT(14H SECOND SPIRAL)
		 CA=AA+TS
		 I=N1
		 J=N2
		 N1=N3
		 N2=N4
		 BA=AA
		 STTS=TTS
		 TTS=TTS1
		 A=ATS
		 CALL RADDEG
		 EE1= XLT
		 EE2= XST
		 GO TO 32
	38	 STS1=E1-STTS
		 STC1=STS1+SAVE
		 N1=I
		 N2=J
		 ZNEW=XY(N2,X)+E3*COS(CB)
	CALL XYOUT(N5,X,ZNEW)
		 ZNEW=XY(N2,Y)-E3*SIN(CB)
	CALL XYOUT(N5,Y,ZNEW)
		 XLT= E2*(3.1415926-CV)
		 TTS= E2*COS(CV/2.)/SIN(CV/2.)
		 ZNEW=XY(N2,X)+TTS*SIN(CB)
	CALL XYOUT(NP,X,ZNEW)
		 ZNEW=XY(N2,Y)+TTS*COS(CB)
	CALL XYOUT(NP,Y,ZNEW)
		 A=CV
		 CALL ANGRAD
		 STC2=STC1+XLT
		 STS2=STC2+OFF
	YY(1)=XY(N1,Y)
	XX(1)=XY(N1,X)
	YY(2)=XY(N2,Y)
	XX(2)=XY(N2,X)
	YY(3)=XY(N4,Y)
	XX(3)=XY(N4,X)
	WRITE(IP,60) XLT,TTS,JD,JM,B,N1,STS1,YY(1),XX(1),N2,STC1
     1,YY(2),XX(2),N4,STC2,YY(3),XX(3)
	60	 FORMAT(' CURVE LENGTH=',F10.3,',TANGENT/CURVE=',F10.3,
     1',VERTEX=',2I3,F7.3/' T.S.=',I3,',STA=',F10.3,', N=',F10.3,
     2', E=',F10.3/' S.C.=',I3,',STA=',F10.3,', N=',F10.3,
     3', E=',F10.3/' C.S.=',I3,',STA=',F10.3,', N=',F10.3,
     4', E=',F10.3)
	YY(1)=XY(N3,Y)
	XX(1)=XY(N3,X)
	YY(2)=XY(N5,Y)
	XX(2)=XY(N5,X)
	YY(3)=XY(NP,Y)
	XX(3)=XY(NP,X)
	WRITE(IP,61)N3,STS2,YY(1),XX(1),N5,YY(2),XX(2),NP,YY(3),XX(3)
61	FORMAT(' S.T.=',I3,',STA=',F10.3,', N=',F10.3,
     5', E=',F10.3/' CENTER OF CIRC CURVE',I4,',N=',F10.3,
     6', E=',F10.3/' VERTEX OF CIRC CURVE',I4,',N=',F10.3,
     7', E=',F10.3)
	CALL CKLIN(7,2)
		 CALL READ
	900	 CALL ERROR(17)
C	 MOD4 CHANGES-L FOR NPI,E3 FOR R, E2 FOR APC, N6 FOR NCURV,
C	 E1 FOB SPC
		 END
