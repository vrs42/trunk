	SUBROUTINE DOREAD
	INTEGER BUF
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA MASK/'@@@@@?'/
		DATA BLANK/1H /
		IF (IS.GE.5) GO TO 25
		IF (IP.NE.4)  WRITE(IP,10)
		IF (IPLOT) 20,11,11
	11	IF ((IP.NE.4).OR.(IS.EQ.4)) GO TO 25
		WRITE(4,13)
	13	FORMAT(' ')
		CALL CKLIN(1,1)
		GO TO 25
	20	WRITE(4,10)
	10    FORMAT(' *'/)
	CALL CKLIN(2,1)
	BUF(3)=0
	ADJFG=0
	NEWVAR=0
	25    READ(IS,1) BFR
	    1 FORMAT(15A6)
6	DO 14 LIMBK=1,15
	LIM=16-LIMBK
	IF (COMP(BFR(LIM),BLANK).NE.1) GO TO 5
	14    CONTINUE
	      LIM=1
5	DO 100 LIMBK=1,6
C	CHARACTER ^ RUBS OUT WHOLE LINE
	CALL CGET(BFR(LIM),LIMBK,II)
	IF ((II.EQ.30).OR.(II.EQ.31)) GO TO 20
100	CONTINUE
		IF (IP.NE.4) GO TO 3
	9	IF (IS.EQ.4) GO TO 8
		IF (IPLOT) 2,8,8
	2	CALL ECHO(BFR,LIM,4)
	CALL CKLIN(1,1)
		GO TO 8
	3	CALL ECHO(BFR,LIM,IP)
		CALL CKLIN(1,2)
	IF(IS.EQ.4)GO TO 8
	IF(IPLOT)1000,8,8
1000	CALL ECHO(BFR,LIM,4)
	CALL CKLIN(1,1)
	8	IF(IS.EQ.4) CALL CKLIN(1,1)
	ZNEW=LSHIFT(BFR(LIM),0,MASK)
	IF (COMP(ZNEW,'      ').NE.1) LIM=LIM+1
	      IF (LIM.GT.15) LIM=15
	      BUF(1)= LIM
	CALL INT0
	      RETURN
