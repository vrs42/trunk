	SUBROUTINE INTSCT(II,JJ)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
		IF (N1.LE.0) CALL ERROR(16)
	      GO TO (3,1,2),II
	 1    CALL ANGRAD
	      YD=A
	      JD=JJD
	      JM=JJM
	      SC=SSC
	      CALL ANGRAD
		IF (N6.LT.0) GO TO 7
	 4    SINB=SIN(YD)
	      COSB=COS(YD)
	      C=YD-A
		IF (SIN(C).EQ.0.) CALL ERROR(28)
      DIST=((XY(N4,X)-XY(N2,X))*COS(A)+(XY(N2,Y)-XY(N4,Y))*SIN(A))
     1/SIN(C)
	CALL XYOUT(N1,X,XY(N2,X)+DIST*SINB)
	CALL XYOUT(N1,Y,XY(N2,Y)+DIST*COSB)
	      GO TO (5,6),JJ
	 5    N=N1
	      CALL PRINT
	      CALL READ
	 6    RETURN
C     BR/INT
	 2    CALL ANGRAD
	      CALL NEWAZ
	      YD=A
	      JD=JJD
	      JM=JJM
	      SC=SSC
	      NQUAD=NNQUAD
	      CALL ANGRAD
	      CALL NEWAZ
	      GO TO 4
C      POINTS/INT
	 3    N=N3
	      J=N2
	      CALL DSTRAD
	      YD= A
	      N=N5
	      J=N4
	      CALL DSTRAD
	      GO TO 4
	7	B= A
		J= N2
		N= N4
		CALL DSTRAD
		YD= YD+A
		A= B+A
		GO TO 4
	      END
