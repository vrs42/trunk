	SUBROUTINE COORD(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
	DATA NCURV,NPC,SPC,NPT,NPI,NC/1,2,3,4,5,6/
C     II=1,COORD/OFFSET
C     II=2,COORD/POA
C     II=3,COORD/EL/POA
		IF (II-2) 1,2,222
	 1    DO 3 KI=1,IA
	      IF(XY(KI,NCURV)-L)3,4,3
	 3    CONTINUE
	      WRITE(1,5)L
	5     FORMAT(' CURVE NO',I5,' NOT STORED')
		CALL CKLIN(1,1)
	      IS= 4
	      CALL READ
	    4 K= KI
	      NN= N
	      CALL RECALC
	      N= NN
	      IF (S-XY(KI,SPC)) 517,517,14
	 517  DIST=XY(KI,SPC)-S
	      A=HBA
	      J=XY(KI,NPC)
	      CALL NEWPT(0)
	 518  A=APC
	      J=N
	523   DIST=-OFF*SIGGN
	 525  CALL NEWPT(0)
	      K=KI-1
	      KCURV= L-1
	      IF (K.LE.0) GO TO 700
	      IF (KCURV.NE.XY(K,NCURV)) GO TO 700
	 11   NN=N
	      CALL RECALC
	      N=NN
	 12   IF(S-SPT)7,30,30
700	ZNEW=XY(KI,NCURV)
	   WRITE(IP,800) ZNEW
	  800 FORMAT(' REQUESTED POINT DOES NOT FALL ON CURVE',I5,
     .', HOWEVER PRINTED RESULTS'/' PERTAIN TO THE BACK TAN'
     .'GENT FOR SPECIFIED CURVE.')
	      CALL CKLIN(2,2)
	      GO TO 30
7	ZNEW=XY(KI-1,NCURV)
	     WRITE(IP,8)  ZNEW
	      CALL CKLIN(2,2)
8     FORMAT(' REQUESTED  POINT CONNECTS WITH CURVE',I5,'. HOWEVER
     1 PRINTED RESULTS'/' PERTAIN TO AN EXTENSION OF BACK TANGENT
     2 FOR SPECIFIED CURVE.')
	 30   GO TO (50,60,60),II
50	ZNEW=XY(KI,NCURV)
	XX(1)=XY(N,X)
	YY(1)=XY(N,Y)
	    WRITE(IP,70) N,S,ZNEW,OFF,YY(1),XX(1)
	      CALL CKLIN(1,2)
	70    FORMAT(' PT',I4,',STA=',F9.3,',CURVE',I5,',OFF=',F9.3,
     1',N=',F9.3,',E=',F9.3)
	      CALL READ
60	ZNEW=XY(KI,NCURV)
	XX(1)=XY(N,X)
	YY(1)=XY(N,Y)
		WRITE(IP,80) N,S,ZNEW,YY(1),XX(1)
	      CALL CKLIN(1,2)
	80    FORMAT(' PT',I4,',STA=',F9.3,',CURVE',I5,',N=',F9.3,
     1',E=',F9.3)
	      IF(II-2)400,400,401
	 400  CALL READ
	 401  CALL VERPT(6)
	      CALL READ
	 14   IF(S-SPT)524,521,521
C     ON AHEAD TANGENT
	 521  DIST=S-SPT
	      J=XY(K,NPT)
	      A=HAA
	      CALL NEWPT(0)
	 522  A=APT
	      J=N
	      DIST=-OFF*SIGGN
	      CALL NEWPT(0)
	      K= KI+1
	      KCURV= L+1
	      IF (K.GT.50) GO TO 900
	      IF (KCURV.NE.XY(K,NCURV)) GO TO 900
	 15   IF(S-XY(KI+1,SPC))30,30,9
900	ZNEW=XY(KI,NCURV)
	   WRITE(IP,1000) ZNEW
	 1000 FORMAT(' REQUESTED POINT DOES NOT FALL ON CURVE',I5,
     .', HOWEVER PRINTED RESULTS'/' PERTAIN TO THE FORWARD'
     .' TANGENT FOR SPECIFIED CURVE.')
	      CALL CKLIN(2,2)
	      GO TO 30
9	ZNEW=XY(KI+1,NCURV)
	     WRITE(IP,10)  ZNEW
	      CALL CKLIN(2,2)
10    FORMAT(' REQUESTED POINT CONNECTS WITH CURVE',I5,'. HOWEVER
     1 PRINTED RESULTS'/' PERTAIN TO AN EXTENSION OF FORWARD TANG
     2ENT FOR SPECIFIED CURVE.')
	      GO TO 30
C     ON CURVE
	524   A=APC+((S-XY(KI,SPC))*SIGGN)/R
	      DIST=R-OFF*SIGGN
	      J=XY(KI,NC)
	      CALL NEWPT(0)
	      GO TO (50,60,60), II
	222	II= 3
	 2    OFF=0.0
	      GO TO 1
		END
