C	IW=0 FOR INPUT,1 FOR OUTPUT;ITYPE=0 FOR ALL,1 FOR POINTS
C	2 FOR DIST,3 FOR ANGLE,4 FOR AREA,5 FOR CURVE
C	IFILE=0 FOR SMALL, 1 FOR LARGE.

	SUBROUTINE TABLES(IW,IQ,ITYPE,IFILE)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	WRITE(2'XYPOS)ARRAYX
	REWIND IQ
	IF((ITYPE.LT.0).OR.(ITYPE.GT.5)) CALL ERROR(15)
	IT1=0
	IF(ITYPE.LT.2) GO TO 1
	IT3=4
	IT1=24
	IF(ITYPE.EQ.5) GO TO 2
	IT1=35-ITYPE
	IT3=1
	IF(IFILE.EQ.1) GO TO 2
	IT3=3
	IT1=31
	GO TO 3
1	IT3=34-10*ITYPE
	GO TO 3
2	IF(IFILE.EQ.0) GO TO 3
	DO 7 IT2=1,IT1
	READ(IQ) ARRAYX
7	CONTINUE
3	IF(IW.EQ.1) GO TO 4
	DO 10 IT2=1,IT3
	IT1=IT1+1
	READ(IQ)ARRAYX
	WRITE(2'IT1)ARRAYX
10	CONTINUE
	IF(ITYPE.EQ.5.OR.ITYPE.EQ.0) IA=ARRAYX(85)
	GO TO 21
4	DO 11 IT2=1,IT3
	IT1=IT1+1
	READ(2'IT1) ARRAYX
	IF(ITYPE.EQ.0.OR.ITYPE.EQ.5)GO TO 100
	GO TO 22
100	IF(IT2.NE.IT3)GO TO 22
	ARRAYX(85)=IA
22	WRITE(IQ)ARRAYX
11	CONTINUE
	IF(IFILE.EQ.0) GO TO 21
	IT1=39-IT1
	DO 20 IT2=1,IT1
	READ(IQ) ARRAYX
20	CONTINUE
21	READ(2'1) ARRAYX
	XYPOS=1
	CALL READ
	END
