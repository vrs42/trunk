	SUBROUTINE DIR2(KOM)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	GO TO (225,226,227,227,229,229,229,232,232,232,235,
     1235,235),(KOM-24)
	225	CALL CONVRT(J,-1)
		CALL CONVRT(N2,-1)
		CALL CONVRT(M,0)
		CALL CONVRT(N,-1)
		CALL CONVRT(K2,-1)
		CALL GRDLTH
	226	CALL CONVRT(N1,0)
		CALL CONVRT(NB,-1)
		CALL CONVRT(N2,-1)
		CALL CONVRT(NA,-1)
		CALL CONVRT(N4,0)
		CALL CONVRT(N3,0)
		CALL CONVRT(N5,0)
		CALL CONVRT(R,1)
		CALL CONVRT(T,1)
		CALL CONVRT(SB,1)
		CALL CONVRT(XL,1)
		CALL ALIGN(1)
C   260 MAY ENTER @ 227 GOING TO COORD(3)
	227	CALL CONVRT(N,0)
		CALL CONVRT(S,1)
		IF (KOM.EQ.27) CALL CONVRT(OFF,1)
		CALL CONVRT(L,0)
		CALL COORD(KOM-26)
	229	CALL CONVRT(N1,0)
		CALL CONVRT(N2,-1)
		II= 2*(KOM-29)
		IF (II.EQ.0) GO TO 230
		CALL GETANG(II-1)
		CALL CONVRT(N4,-1)
		CALL GETANG(II)
		GO TO 231
	230	CALL CONVRT(N3,-1)
		CALL CONVRT(N4,-1)
		CALL CONVRT(N5,-1)
	231	CALL INTSCT(II/2+1,1)
	232	CALL CONVRT(N1,0)
		CALL CONVRT(N3,-1)
		CALL CONVRT(XN,1)
		CALL CONVRT(N4,-1)
		IF (KOM-33) 228,233,234
	228	CALL CONVRT(YN,1)
		GO TO 224
	233	CALL CONVRT(N5,-1)
		GO TO 224
	234	CALL GETANG(1)
	224	CALL CONVRT(N2,-1)
		CALL ARC(KOM-31)
	235	CALL CONVRT(N3,0)
		CALL CONVRT(NB,-1)
		CALL GETANG(1)
		IF (KOM.NE.37) GO TO 236
		CALL CONVRT(T,1)
		CALL CONVRT(E1,1)
	236	CALL CONVRT(E3,1)
		IF (KOM.EQ.37) GO TO 237
		CALL CONVRT(XL,1)
		CALL CONVRT(E2,1)
	237	CALL CONVRT(N4,-1)
		IF (KOM.NE.36) GO TO 336
		CALL CONVRT(DIST,1)
		CALL CONVRT(N1,-1)
		GO TO 337
	336	CALL GETANG(2)
	337	IF (KOM.NE.37) CALL SPIRAL(KOM-34)
		CALL CONVRT(OFF,1)
		CALL CONVRT(E4,1)
		CALL CONVRT(BA,1)
		CALL SPISPI
	END
