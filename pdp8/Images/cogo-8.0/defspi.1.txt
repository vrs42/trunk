	SUBROUTINE DEFSPI(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA NCURV,NPC,SPC,NPT,NPI,NC/1,2,3,4,5,6/
		GO TO (1,2),II
	1	IA= IA+1
		IF (IA.LE.50) GO TO 570
		WRITE(4,500)N1
	500	FORMAT(' AN ATTEMPT TO DEFINE MORE THAN 50 CURVES HAS'
     1,' BEEN MADE'/' NO VALUES HAVE BEEN STORED FOR CURVE',I5)
		CALL CKLIN(2,1)
		IS= 4
		CALL READ
	570	IPRES= IA
	CALL XYOUT(IA,NCURV,N1)
	CALL XYOUT(IA,NPI,N2)
	CALL XYOUT(IA,NPC,N4)
	CALL XYOUT(IA,NPT,N5)
	CALL XYOUT(IA,SPC,E1)
	2	J= N4
		N= N2
		CALL DSTRAD
		APC= A
		DIFF= DIST
		J= N2
		N= N5
		CALL DSTRAD
		APT= A
		A= A-APC
		IF (A.GT.3.1415926) A=A-6.2831853
		IF (A.LT.-3.1415926) A= A+6.2831853
		IF (SIGGN*A.LT.0.) CALL ERROR(5)
		IF (II.EQ.1) R= R*SIGGN
		IF (E2.NE.0.) E3= E2-E1
		IF (E3.NE.0.) E2= E1+E3
		IF (R.NE.0.) GO TO 3
		IF ((E2.EQ.0.).AND.(E3.EQ.0.)) CALL ERROR(3)
		R= E3/2./A
		GO TO 4
	3	E3=R*2.*A
		E2= E1+E3
		IF (II.EQ.2) RETURN
4	CALL XYOUT(IA,NC,R)
		CALL RADDEG
		JJD= JD
		JJM= JM
		SSC= B
		A=APC
		CALL RADDEG
		JD1= JD
		JM1= JM
		SC1= B
		A=APT
		CALL RADDEG
		WRITE(IP,100) N1,E3,JJD,JJM,SSC,DIFF,JD1,JM1,SC1,DIST
     1,JD,JM,B
 	100	FORMAT(' SPIRAL',I5,': L=',F9.3,', DEFLN ANG=',
     1 2I4,F5.1/' LONG TAN=',F8.3,', AZ=',2I4,F5.1/
     1 ' SHORT TAN=',F8.3,', AZ=',2I4,F5.1)
		CALL CKLIN(3,2)
		CALL READ
		END
