	FUNCTION XY(V,P)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	INTEGER XYPOS
	IF (P.GT.0) GO TO 8
	IF (V.GE.1000) GO TO 1
	IMX=IFIX((V-1)/42)+1
	IPX=2*(V-(IMX-1)*42)+P+1
5	IF (IMX.EQ.XYPOS) GO TO 2
	WRITE(2'XYPOS)ARRAYX
	READ(2'IMX)ARRAYX
	XYPOS=IMX
2	XY=ARRAYX(IPX)
	RETURN
1	IPX=(V-999)*2+P+1
	XY=XY3(IPX)
	RETURN
C	V IS THE ELEMENT POSITION AND P IS THE BLOCK
C	POINTER. 1-6 FOR CURVE,7-11 FOR PI'S,12-17
C	FOR VERT CROSS SEC., 19 FOR ASAVE
C	20 FOR DSAVE, 18 FOR RSAVE,21 FOR IE
8	IF(P.LT.21)GO TO 3
	IMX=IFIX((V-1)/85)+35
	IPX=V-(IMX-35)*85
	GO TO 5
3	IF (P.LT.18) GO TO 4
	IMX=P+14
	IPX=V
	GO TO 5
4	IF (P.GT.6) GO TO 6
	IMX=IFIX((V-1)/14)+25
	IPX=6*((V-1)-(IMX-25)*14)+P
	GO TO 5
6	IF (P.GT.11) GO TO 7
	IMX=IFIX((V-1)/17)+29
	IPX=5*((V-1)-(IMX-29)*17)+P-6
	GO TO 5
7	IMX=31
	IPX=6*(V-1)+P-11
	GO TO 5
	END
