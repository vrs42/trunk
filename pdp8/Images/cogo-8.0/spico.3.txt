	SUBROUTINE SPICO(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA NCURV,NPC,SPC,NPT,NPI,NC/1,2,3,4,5,6/
	DATA X,Y/-2,-1/
		DO 1 I=1,IA
		IF (XY(I,NCURV).EQ.L) GO TO 2
1		CONTINUE
		CALL ERROR(19)
	2	IPRES= I
	N2=XY(I,NPI)
	N4=XY(I,NPC)
	N5=XY(I,NPT)
	E1=XY(I,SPC)
	R=XY(I,NC)
		SIGGN= SIGN(1.,R)
		N1= NP
		N3= J
		CALL DEFSPI(2)
		SINA= SIN(APC)
		COSA= COS(APC)
		T1= 1/2./R/E3
		IF (II.EQ.1) GO TO 4
	CALL XYOUT(N1,X,XY(N4,X))
	CALL XYOUT(N1,Y,XY(N4,Y))
		AA= APC
		A= 0.
		XL= 0.
		DO 3 IJ=1,35
		A= APC+A
		AA= A+1.5707963
	DIST= (XY(N3,Y)-XY(N1,Y))*SIN(AA)+(XY(N1,X)-XY(N3,X))*COS(AA)
		IF (ABS(DIST).LE..001) GO TO 6
		XL= XL+DIST
		IF (XL.LE.0.) GO TO 5
		IF (XL.GT.(E3+.01)) GO TO 12
	17	T3= XL*T1*XL/3.*XL
		T2= .9*T3*T3/XL
		T4= 5.*T2*T3/XL/7.
		XN= T3-T4
		YN= XL-T2
		ZNEW=XY(N4,X) +YN*SINA+XN*COSA
	CALL XYOUT(N1,X,ZNEW)
		ZNEW=XY(N4,Y)+YN*COSA-XN*SINA
	CALL XYOUT(N1,Y,ZNEW)
		IF (II.EQ.1) GO TO 6
	3	A= XL*XL*T1
		A= AA-APC-A
		CALL RADDEG
		WRITE(IP,1030) JD,JM,B,N1,N3
	1030	FORMAT(' OFFSET ANGLE=',2I4,F4.0,' FROM',I4,' TO',I4)
		CALL CKLIN(1,2)
		GO TO 6
	5	IS= 4
		WRITE(4,1000)N3
	1000	FORMAT(' PT',I4,' IS OFFSET FR TAN.  USE "TOF"')
		CALL CKLIN(1,1)
		CALL READ
	4	XL= S-E1
		IF (XL.LE.0.) GO TO 5
		IF (XL.LE.E3) GO TO 17
	12	IS= 4
		WRITE(4,1001)N3
	1001	FORMAT(' PT',I4,' IS OFFSET FROM ARC.  USE "OAL"')
		CALL CKLIN(1,1)
		CALL READ
	6	N= N1
		CALL PRINT
		IF (II.EQ.1) CALL READ
		A= E1+XL
		WRITE(IP,1007) N,A
	1007	FORMAT(' PT',I4,' AT STA',F10.3)
		CALL CKLIN(1,2)
		J= N3
		CALL DISTAN(1)
		END
