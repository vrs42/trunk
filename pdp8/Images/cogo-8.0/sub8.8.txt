	TADZ	GOBACK+1

	DCAZ	ARG3
	JMS	GET
	DCAZ	ARG2
	TADZ	ARG3
	DCA	CDF2

	DCAZ	ARG3

/LSHIFT ROUTINE
LSHF8,	0
	CLA
	TADZ IX1		/GET SHIFT COUNT
	SMA
	JMP LSH2
	DCAZ PONTX		/SHIFT LEFT
	TADZ S0101
	DCAZ TEMP	/BACKWARD 4040
	JMP LSH1

LSH2,	CIA
	DCAZ PONTX
	TADZ S4040
	DCAZ TEMP	/SPACE FILL IN

LSH1,	CLL CLA IAC	/GET SPACE BIT
	ANDZ TEMP
	SZA CLA
	CML
	TADZ TEMP	/ROTATE BIT OUT AND BACK IN
	RAR
	DCAZ TEMP

	TADZ IX1		/TEST LEFT OR RIGHT SHIFT
	SPA CLA
	JMP LSH3
	JMS ROTATE
	ADDR ARG1
	JMP LSH4

LSH3,	JMS ROTAL	/ROTATE LEFT
	ADDR ARG1+2

LSH4,	ISZZ	PONTX	/GET NEXT VALUE
	JMP	LSH1
	CLL CLA
	CDF CIF 0
	JMP% LSHF8
ROTATE,	0	/ROTATE RIGHT ROUTINE
	ISZ ROTATE	/SKIP FIELD ADD.
	TAD% ROTATE	/GET ADD OF 36 BIT WORD
	DCAZ LITEX
	ISZ ROTATE
	TADZ MIN3
	DCAZ TEMP1
ROTAT1,	TADZ% LITEX	/ROTATE 36 BIT WORD  1 BIT TO RIGHT
	RAR
	DCAZ% LITEX
	ISZZ LITEX
	ISZZ TEMP1
	JMP ROTAT1
	JMP% ROTATE

ROTAL,	0	/ROTATE A 36 BIT WORD 1 BIT LEFT
	ISZ ROTAL
	TAD% ROTAL	/GET ADD OF WORD
	DCAZ LITEX
	ISZ ROTAL
	TADZ MIN3
	DCAZ TEMP1
ROTAL1,	TADZ% LITEX	/START FROM BOTTOM UP
	RAL
	DCAZ% LITEX	/LINK IS SET
	CLA CMA
	TADZ LITEX	/COMPLEMENTS THE LINK
	DCAZ LITEX
	CML		/RESET THE LINK
	ISZZ TEMP1
	JMP ROTAL1
	JMP% ROTAL
/SLITE AND SLITET CODE

	ORG COMPS+200
COD8,	0	/SUB TO SET SENSE LITES
	CLA
	TADZ IX1		/GET SENSE LITE #
	SZA
	JMP COD8A
	DCAZ LITE	/0 ALL LITES ON 0
	DCAZ LITE+1
	DCAZ LITE+2
	CDF CIF 0
	JMP% COD8

COD8A,	DCAZ PONTX	/SAVE LITE #
	JMS CODS	/GET BIT POINTER AND WORD POINTER
	TADZ PONTX	/HAS BIT POINTER
	CMA
	ANDZ% LITEX
	TADZ PONTX
	DCAZ% LITEX	/SET LITE
	CDF CIF 0
	JMP% COD8

/SLITET CODE
COD8T,	0		/SLITET ,READ SENSE LITE AND CLEAR
	CLA
	TADZ IX1
	DCAZ PONTX	/STORE LITE #
	JMS CODS	/GET BIT AND WORD POINTER
	TADZ PONTX	/HAS BIT POINTER
	ANDZ% LITEX	/TEST IF LIGHT IS ON
	SNA CLA
	IAC
	IAC
	DCAZ IX0		/SET ANS=1 IF ON,2 IF OFF
	TADZ PONTX	/CLEAR LITE
	CMA
	ANDZ% LITEX
	DCAZ% LITEX
	CDF CIF 0
	JMP% COD8T

/LSHIFT MASK ROUTINE
LSHMK8,	0
	CLA CLL
	TADZ ARG2
	TADZ ARG2+1
	TADZ ARG2+2
	SNA CLA		/TEST FOR 0 MASK
	JMP LSK1
	TADZ ARG1	/AND TWO ARGS,ARG1.AND.MASK
	ANDZ ARG2
	DCAZ ARG1
	TADZ ARG1+1
	ANDZ ARG2+1
	DCAZ ARG1+1
	TADZ ARG1+2
	ANDZ ARG2+2
	DCAZ ARG1+2
	TADZ ARG2	/C(MASK).AND.SPACES
	CMA
	ANDZ S4040	/ARG3=TAD OF BOTH ANS.
	TADZ ARG1
	DCAZ ARG1
	TADZ ARG2+1
	CMA
	ANDZ S4040
	TADZ ARG1+1
	DCAZ ARG1+1
	TADZ ARG2+2
	CMA
	ANDZ S4040
	TADZ ARG1+2
	DCAZ ARG1+2
LSK1,	CDF CIF 0
	JMP% LSHMK8
/SUB TO LOCATE BIT AND WORD OF 36 BIT WORD FOR SLITE,SLITET
CODS,	0	/ENTER WITH LITE # IN PONTX
	TADZ TEMP1+1
	DCAZ LITEX	/ADD OF LITE
COD2,	TADZ PONTX
	TADZ MIN14	/TEST IF LITE IN WORD
	SPA SNA
	JMP COD1
	ISZZ LITEX	/NO,TEST NEXT WORD
	DCAZ PONTX
	JMP COD2

COD1,	TADZ POS14	/GET POINTER TO LITE ADD.
	CIA
	DCAZ PONTX	/SET COUNTER
	CLA CLL CML	/ROTATE BIT TO LITE BIT
COD3,	RAL
	ISZZ PONTX
	JMP COD3
	DCAZ PONTX
	JMP% CODS
LORS,	0		/LOR SUBROUTINE
	CLA CLL
	TADZ ARG1+2
	AND LORS1	/7700
	DCAZ ARG1+2
	TADZ ARG2+2
	AND LORS2
	TADZ ARG1+2
	DCAZ ARG1+2
	CDF CIF 0
	JMP% LORS
LORS1,	7700
LORS2,	0077

	TRAP4 LSHF8	/CALL LSHIFT ROUTINE
