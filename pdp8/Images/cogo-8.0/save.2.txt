	SUBROUTINE SAVE(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA RSAVE,ASAVE,DSAVE/18,19,20/
		II= II-10
		IF (N1.GT.50) CALL ERROR(20)
		GO TO (1,2,3,4,5,6,7,7,5), II
1	ZNEW=XL
		GO TO 90
2	ZNEW=YN
		GO TO 90
3	ZNEW=R
		GO TO 90
4	ZNEW=T
90	CALL XYOUT(N1,DSAVE,ZNEW)
	9	 WRITE(IP,100) N1,ZNEW
	100	FORMAT(' DISTANCE('I2,')= ',F10.3)
	99	CALL CKLIN(1,2)
		CALL READ
	5	IF (A.LE.6.2831853) GO TO 10
		A= A-6.2831853
		GO TO 5
	10	IF (A.GT.0.) GO TO 15
		A= A+6.2831853
		GO TO 10
15	CALL XYOUT(N1,ASAVE,A)
		IF (II.EQ.8) GO TO 8
		CALL RADDEG
		 WRITE(IP,101) N1,JD,JM,B
	101	FORMAT(' ANGLE(',I2,')= ',I4,I3,F7.3)
		GO TO 99
6	CALL XYOUT(N1,RSAVE,AREA)
		 WRITE(IP,102) N1,AREA
	102	FORMAT(' AREA(',I2,')= ',F12.3)
		GO TO 99
	7	CALL DSTRAD
		GO TO 5
	8	CALL QUAD
		CALL RADDEG
		CALL ALPHA(ALIB,BLIB)
		WRITE(IP,103) N1,ALIB,JD,JM,B,BLIB
	103	FORMAT(' ANGLE(',I2,')= BRNG ',A2,2I3,F7.3,1X,A2)
		GO TO 99
		END
