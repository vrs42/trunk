	SUBROUTINE ALIGN(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA NCURV,NPC,SPC,NPT,NPI,NC/1,2,3,4,5,6/
C     ALIGNMENT
C     COMPUTE CURVE ELEMENTS ROUTINE
C     ENTER WITH NB,NPI,NA,NPC,NC,NPT
C     R,T,SB,XL * IF R=0., COMPUTE FROM T, IF T=0., COMPUTE FROM R,
C     IF BOTH ARE ZERO, COMPOUND
C     GET BA BACK AZIMUTH
	      GO TO (5,6),II
	 5    IA=IA+1
	      IF (IA.GT.50) GO TO 530
	      IPRES=IA
	CALL XYOUT(IA,NCURV,N1)
	CALL XYOUT(IA,NPI,N2)
	CALL XYOUT(IA,NPC,N4)
	CALL XYOUT(IA,NPT,N5)
	CALL XYOUT(IA,NC,N3)
	 6    J=N2
	      N=NB
	      CALL DSTRAD
	      HBA=A
	      TT=DIST
C     GET AA AHEAD AZIMUTH
	      N=NA
	      CALL DSTRAD
	      HAA=A
C     COMPUTE VERTEX ANGLE AND SIGN
	      VA=HBA-HAA
	      E1=ABS(VA)
	      SIGGN = 1.0
	      IF(E1-3.1415927)538,538,535
	  530 WRITE(4,500)N1
	  500 FORMAT(' AN ATTEMPT TO DEFINE MORE THAN 20 CURVES HAS'
     .' BEEN MADE'/' NO VALUES HAVE BEEN STORED FOR CURVE',I5)
	      CALL CKLIN(2,1)
	      IS= 4
	      CALL READ
	 535  IF(VA)537,537,536
	536   SIGGN=-1.0
	 537  DELTA=-3.1415927+E1
	      GO TO 600
	 538  IF(VA)539,541,541
	539   SIGGN=-1.0
	 541  DELTA=3.1415927-E1
	 600  HALF=DELTA/2.0
	      GO TO (1,2),II
C     COMPUTE T OR R OR BOTH AND STATION PC AND PT
	 1    IF(R+T)545,545,542
	 542  IF(R)544,544,543
	 543  T=R*SIN(HALF)/COS(HALF)
	      GO TO 555
	 544  R=T*COS(HALF)/SIN(HALF)
	 555  XL=TT-T
	      IF(SB)553,554,554
553	CALL XYOUT(IA,SPC,SPT+XL)
	      GO TO 556
	 554  SPT=SB
	      GO TO 553
	 545  T=TT-XL
	      GO TO 544
	 556  XCUR=R*DELTA
	      SPT=XY(IA,SPC)+XCUR
C     PRINT CURVE ELEMENTS
	      A=DELTA
	      CALL RADDEG
	552   NSIGN=SIGGN
	ZNEW1=XY(IA,NCURV)
	ZNEW2=XY(IA,SPC)
	      WRITE(IP,52) ZNEW1,R,NSIGN,T,JD,JM,B,XCUR,XL,ZNEW2,SPT
	52    FORMAT(' NO=',I5,' RAD=',F10.3,', SIGN=',I2,', TAN=',
     1F10.3,', DEFL=',2I3,F7.3/' CURVE=',F10.3,', TAN SECT=',
     2F10.3,', PC=',F10.3,', PT=',F10.3)
	      CALL CKLIN(2,2)
C     COMPUTE AZ OF PC APC
	2     APC=HBA+1.5707963*SIGGN
	      IF(APC)547,548,548
	 547  APC=APC+6.2831853
	 548  IF(APC-6.2831853)551,549,549
	 549  APC=APC-6.2831853
	551   APT=HAA-1.5707963*SIGGN
	      GO TO (3,4),II
	 4    RETURN
C     LOCATE PC,PT, AND CENTER
	 3    N=XY(IA,NPC)
	      J=XY(IA,NPI)
	      DIST=T
	      A=HBA
	      CALL NEWPT(1)
	      N=XY(IA,NPT)
	      A=HAA
	      CALL NEWPT(1)
	      N=XY(IA,NC)
	      J=XY(IA,NPC)
	      DIST=-R
	      A=APC
	      CALL NEWPT(1)
	      CALL READ
	      END
