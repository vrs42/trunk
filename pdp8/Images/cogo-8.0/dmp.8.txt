	SUBROUTINE DMP
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	      DATA FILL/'_? @@A'/
	DATA Y/-1/,X/-2/
C     DUMP ROUTINE
	DATA NCURV,NPC,SPC,NPT,NPI,NC/1,2,3,4,5,6/
	DATA RSAVE,ASAVE,DSAVE/18,19,20/
	IF((I.LE.0).OR.(I.GT.5)) CALL ERROR(16)
	IF(N.LE.0) CALL ERROR(16)
	GO TO (19,21,22,23,24),I
21	WRITE(IP,31)
31	FORMAT(' NUM',10X,'DIS')
	GO TO 20
22	WRITE(IP,32)
32	FORMAT(' NUM',5X,'ANGLE')
	GO TO 20
23	WRITE(IP,33)
33	FORMAT(' NUM',9X,'AREA')
	GO TO 20
24	WRITE(IP,34)
34	FORMAT(' NSECT NPC',8X,'SPC NPI NPT  NC')
	GO TO 20
19	WRITE(IP,35)
35	FORMAT('  PT',5X,'N',9X,'E')
20	CALL CKLIN(1,2)
	DO 10 II=N,J
	GO TO(1,2,4,3,5),I
1	XX(1)=XY(II,X)
	IF(XX(1).EQ.FILL) GO TO 10
	YY(1)=XY(II,Y)
	WRITE(IP,43)II,YY(1),XX(1)
43	FORMAT(1H ,I3,2F10.3)
	GO TO 9
2	XX(1)=XY(II,DSAVE)
41	IF(XX(1).EQ.0.) GO TO 10
	WRITE(IP,30)II,XX(1)
30	FORMAT(1H ,I3,F13.3)
	GO TO 9
3	XX(1)=XY(II,RSAVE)
	GO TO 41
4	A=XY(II,ASAVE)
	IF(A.EQ.0.) GO TO 10
	CALL RADDEG
	WRITE(IP,44)II,JD,JM,B
44	FORMAT(1H ,I3,2I4,F7.3)
	GO TO 9
5	XX(1)=XY(II,NCURV)
	IF(II.GT.IA) GO TO 10
	XX(2)=XY(II,NPC)
	XX(3)=XY(II,SPC)
	YY(1)=XY(II,NPI)
	YY(2)=XY(II,NPT)
	YY(3)=XY(II,NC)
	WRITE(IP,42)XX,YY
42	FORMAT(1H ,I5,I4,F11.3,3I4)
9	CALL CKLIN(1,2)
10	CONTINUE
	CALL READ
	END
