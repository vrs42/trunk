	SUBROUTINE SPIRAL(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
		IF (N3.LE.0) CALL ERROR(16)
	      ITEST=0
	      XTEST=XL
	      GO TO (1,2),II
	 1    N1=1000
	      N2=NB
	      CALL INTSCT(2,2)
	      AA=A
	      N=1000
	      J=N4
	      CALL DSTRAD
	      D=DIST
	 4    J=NB
	      CALL DSTRAD
	      XL=DIST
	    8 T1= 1./(2.*E2*XTEST)
	XS=XY(NB,X)
	YS=XY(NB,Y)
	XP=XY(N4,X)
	YP=XY(N4,Y)
		IJ= 1
	19    SINA=SIN(AA)
	      COSA=COS(AA)
	      GO TO (5,6),II
	 5    PF1D=-SINA
	      PF2D=-COSA
	      GO TO 7
	 6    PF1D=-D*COSA
	      PF2D=D*SINA
	 7    SINB=SIN(YD)
	      COSB=COS(YD)
		DO 16 KT=1,35
	      T2=(XL**5*T1**2)/10.0
	      T3=(XL**3*T1)/3.0
	      T4=(XL**7*T1**3)/42.0
	      T8=1.0-(T1**2*XL**4)/2.0
	      T9=T1*XL**2-(T1**3*XL**6)/6.0
	       T5=XL-T2
	      T6=T3-T4
	      F1 = T5*SINB+T6*COSB+XS-XP-D*SINA
	      F2 = T5*COSB-T6*SINB + YS-YP-D*COSA
	       PF1L=T8*SINB+T9*COSB
	      PF2L=T8*COSB-T9*SINB
	       DELL=(PF1D*F2-PF2D*F1)/(PF1L*PF2D-PF1D*PF2L)
	      DELD=-(PF2L*DELL+F2)/PF2D
		IF (SQRT(F1*F1+F2*F2).LT..001) GO TO 12
	 13   GO TO (14,15),II
	 14   D=D+DELD
	      GO TO 16
	 15   AA=AA+DELD
	      SINA=SIN(AA)
	      COSA=COS(AA)
	      PF1D=-D*COSA
	      PF2D=D*SINA
	 16   XL=XL+DELL
		GO TO (25,41), JJ
	 12   S=E3+XL
	CALL XYOUT(N3,X,XP+D*SINA)
	CALL XYOUT(N3,Y,YP+D*COSA)
	      GO TO (10,11),II
	10	IF ((XTEST.LT.XL).OR.(XL.LT.0.)) CALL ERROR(28)
	XX(1)=XY(N3,X)
	YY(1)=XY(N3,Y)
	      WRITE(IP,20) N3,YY(1),XX(1),S,XL,D
	20    FORMAT(' PT=',I4,', N=',F10.3,', E=',F10.3,', STA=',F10.3
     1/' SPIRAL (T.S. TO N)=',F10.3,', DIST (NP TO N)=',F10.3)
	      CALL CKLIN(2,2)
	      CALL READ
	11    GO TO(21,22),JJ
	 21   E1=XL
	50	IF ((XTEST.LT.XL).OR.(XL.LT.0.)) GO TO 25
23	XY3(3)=XY(N3,X)
	XY3(4)=XY(N3,Y)
	      E4=S
	      E5=D
	      GO TO 24
	 25   ITEST=1
	 24   JJ=2
	      J=N4
	      N=NB
	      CALL DSTRAD
	      AA=A
	      XL=0.0
	      GO TO 8
	22	IF ((XTEST.GE.XL).AND.(XL.GE.0.)) GO TO 42
	 41   IF(ITEST)44,43,44
	 43   NP=N3
	CALL XYOUT(NP,X,XY3(3))
	CALL XYOUT(NP,Y,XY3(4))
	      S=  E4
	      D=E5
	      XL=E1
100	XX(1)=XY(N3,X)
	YY(1)=XY(N3,Y)
	   WRITE(IP,30) N3,YY(1),XX(1),S,XL,D
	30    FORMAT(' PT=',I4,', N=',F10.3,', E=',F10.3,', STA=',F10.3
     1/' SPIRAL (T.S. TO N)=',F10.3,', RADIUS OF ARC=',F10.3)
	      CALL CKLIN(2,2)
	      CALL READ
	44    WRITE(4,110)
 110  FORMAT(49H THERE IS NO INTERSECTION OF THE CURVE AND SPIRAL)
	      CALL CKLIN(1,1)
	      CALL READ
	 42   IF(ITEST)100,45,100
	 45   J=N3
	      N=N1
	      CALL DSTRAD
	      DIST1=DIST
	      J=1001
	      CALL DSTRAD
	      IF(DIST-DIST1)43,100,100
	2     D=DIST
	      JJ=1
	      CALL ANGRAD
	      YD=A
	      J=NB
	      N=1000
	      DIST=XL
	      CALL NEWPT(0)
	      J=N4
	      CALL DSTRAD
	      AA=A
	      GO TO 4
		END
