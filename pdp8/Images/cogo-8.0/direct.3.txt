		SUBROUTINE DIRECT(KOM)
1	IF(KOM.LE.0) CALL DIR3(KOM)
	IF((KOM.EQ.101).OR.(KOM.EQ.83))GO TO 10
	IF((KOM.GE.92).AND.(KOM.LE.98))GO TO 10
	IF((KOM.GT.0).AND.(KOM.LE.24))CALL DIR1(KOM)
	IF((KOM.GT.24).AND.(KOM.LE.37))CALL DIR2(KOM)
	IF((KOM.GT.37).AND.(KOM.LE.54))CALL DIR3(KOM)
	IF((KOM.GT.54).AND.(KOM.LT.83))CALL DIR4(KOM)
	IF((KOM.GT.83).AND.(KOM.LT.92))CALL DIR5(KOM)
	IF((KOM.GT.98).AND.(KOM.LE.110))CALL DIR6(KOM)
	RETURN
10	CALL DIR7(KOM)
	GO TO 1
	END
