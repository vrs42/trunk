	SUBROUTINE ADJUST(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	COMMON/OOO/SDXDY,SDXDX,SDYDY,SLAT,SDEP
	DATA X,Y/-2,-1/,IE/21/
	      GO TO (1,109,109),II
1	E1=XY(N6,X)
	E2=XY(N6,Y)
	      E3=0.0
	      SDXDY=0.0
	      SDXDX=0.0
	      SDYDY=0.0
		SLAT= 0.
		SDEP= 0.
	      I=1
	CALL XYOUT(1,IE,N1)
	      CALL READ
	 109  I=I+1
	      GO TO (900,320,130),II
	 900  CALL ERROR(14)
	 320  CALL ANGRAD
		IF (N4.GT.0) CALL NEWAZ
	      CALL NEWPT(0)
101	CALL XYOUT(I,IE,N)
	      E3=E3+DIST
	      SDXDY=SDXDY+DX*DY/DIST
	      SDXDX=SDXDX+DX*DX/DIST
	      SDYDY=SDYDY+DY*DY/DIST
		SLAT= SLAT+ABS(DY)
		SDEP= SDEP+ABS(DX)
	      IF(N-N6)400,116,400
	 400  J=N
	      CALL READ
	 130  BA=A
	      CALL ANGRAD
		IF (N4.GT.0) CALL NEWAZ
	 131  A=A+BA
		IF (N4.EQ.0) A= A-3.1415926
	      CALL NEWPT(0)
	      J=N
	      GO TO 101
116	XY3(1)=XY(N,X)
	XY3(2)=XY(N,Y)
	CALL XYOUT(N6,X,E1)
	CALL XYOUT(N6,Y,E2)
	      N=N6
	      J=1000
	      CALL DSTRAD
	118   WRITE(IP,10) DY,DX,DIST
	10    FORMAT(' ERROR: N=',F9.3,': E=',F9.3,': TOTAL=',F9.3)
	      CALL CKLIN(1,2)
	DIST=E3/DIST
	      CALL RADDEG
	119   WRITE(IP,20) JD,JM,DIST,E3
20    FORMAT(' CLOSURE: AZ=',I3,I3,': ERR=',F9.1,', PERIM=',F9.3)
	      CALL CKLIN(1,2)
	      N=N1
	      E1=0.0
	      E2=0.0
		GO TO (500,600,700), N3
C  TRANSIT RULE
	700	XN= DX/SDEP
		YN= DY/SLAT
		GO TO 41
C  COMPASS RULE
	600	XN= DX/E3
		YN= DY/E3
		GO TO 41
	500      E3=SDXDX*SDYDY-(SDXDY)**2
	      XN=(DX*SDXDY-DY*SDXDX)/E3
	      YN=(DY*SDXDY-DX*SDYDY)/E3
	41      WRITE(IP,40)
	 40   FORMAT('0TABLE OF ADJUSTED COORDINATES'/'0 PT',5X,'N',9X
     1,'E'/)
	      CALL CKLIN(5,2)
	      N2=I-1
		I= XY(1,IE)
	XX(1)=XY(I,X)
	YY(1)=XY(I,Y)
		WRITE(IP,30)I,YY(1),XX(1)
		CALL CKLIN(1,2)
	      DO 120 I=2,N2
	      J=N
	      N=XY(I,IE)
	      CALL DSTRAD
		GO TO (510,610,710), N3
C  TRANSIT ADJUSTMENT
	710	E1= E1+DX*XN
		E2= E2+DY*YN
		GO TO 51
C  COMPASS ADJUSTMENT
	610	E1= E1+DIST*XN
		E2= E2+DIST*YN
		GO TO 51
	510      E1= E1+(XN*DX*DY+YN*DX*DX)/DIST
	      E2= E2+(XN*DY*DY+YN*DX*DY)/DIST
51	XX(1)=XY(N,X)-E1
	CALL XYOUT(N,X,XX(1))
	YY(1)=XY(N,Y)-E2
	CALL XYOUT(N,Y,YY(1))
	      CALL CKLIN(1,2)
	YY(1)=XY(N,Y)
	XX(1)=XY(N,X)
	WRITE(IP,30)N,YY(1),XX(1)
	30	FORMAT(I4,2F12.3)
120	CONTINUE
	      N=N1
	      N2=N2+1
		IF (XY(N2,IE).EQ.XY(1,IE)) GO TO 2
		I= XY(N2,IE)
	XX(1)=XY(I,X)
	YY(1)=XY(I,Y)
		WRITE(IP,30) I,YY(1),XX(1)
		CALL CKLIN(1,2)
	2	WORD= 'AZIM'
		IF (N4.GT.0) WORD= 'BRNG'
		WRITE(IP,50) WORD
 50   FORMAT('0TABLE OF ADJUSTED DISTANCES'/'0 PT TO PT    DIST',
     19X,A4/)
	      CALL CKLIN(5,2)
		ALIB= ' '
		BLIB=  ' '
	      DO 121 I=2,N2
	      J=N
	      N=XY(I,IE)
	      CALL DSTRAD
		IF (N4.GT.0) CALL QUAD
	      CALL RADDEG
		IF (N4.LE.0) GO TO 61
		CALL ALPHA(ALIB,BLIB)
	61      WRITE(IP,60) J,N,DIST,ALIB,JD,JM,B,BLIB
	60	FORMAT(I4,I6,F12.3,1X,A2,2I3,F7.3,1X,A2)
	      CALL CKLIN(1,2)
	 121  CONTINUE
	      CALL READ
C     SUBSTITUTIONS FROM MOD2 E1 FOR APC, E2 FOR APT, E3 FOR R,
C     N6 FOR NPT, N2 FOR NC
	      END
