	SUBROUTINE DIR3(KOM)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	COMMON/NNN/ADJFG
	IF(KOM.LE.0)GO TO 198
	GO TO (238,239,239,241,241,241,244,244,244,244,244,250,250,250
     1,252,253,254),(KOM-37)
238	CALL CONVRT(N6,0)
	CALL CONVRT(NB,-1)
	CALL CONVRT(L,-1)
	CALL CONVRT(NA,-1)
	CALL CONVRT(XL,1)
	CALL CONVRT(E2,1)
	CALL CONVRT(OFF,1)
	CALL CONVRT(E1,1)
	CALL CONVRT(N1,0)
	CALL CONVRT(N2,0)
		CALL CONVRT(N3,0)
	CALL CONVRT(N4,0)
	CALL CONVRT(N5,0)
	CALL CONVRT(NP,0)
	CALL CRVSPI
198	CALL CONVRT(N1,-1)
	CALL CONVRT(N6,-1)
	CALL ADJUST(1)
239	IF (KOM.EQ.39) CALL CONVRT(J,-1)
	CALL CONVRT(N,0)
	CALL CONVRT(DIST,1)
	II= 1
	IF((N4.GT.0).AND.(KOM.EQ.39))II=3
	IF(II.NE.3)GO TO 10
	IDTCT=IDTCT-1
	ADJFG=1
	JD=-999
10	CALL GETANG(II)
	CALL ADJUST(KOM-37)
241	CALL CONVRT(K,-1)
	CALL AREAS(KOM-40)
	NEWVAR=1
199	CALL INTA
244	CALL CONVRT(S,1)
	IF (MOD(KOM,2).NE.1) CALL CONVRT(OFF,1)
	II= KOM-45
	IF (II) 245,246,247
247	IF (II-2) 240,248,249
240	CALL CONVRT(E4,1)
	GO TO 248
245	II= 0
249	CALL CONVRT(E3,1)
	GO TO 248
246	II= -1
	CALL CONVRT(E5,1)
248	CALL VERPT(II+2)
200	CALL READ
250	CALL CONVRT(N,0)
	CALL CONVRT(S,1)
	CALL CONVRT(E1,1)
	IF (KOM.NE.50) GO TO 251
	CALL CONVRT(E2,1)
	CALL CONVRT(E3,1)
251	CALL VERT(KOM-48)
252	CALL CONVRT(DIFF,1)
	GO TO 200
253	CALL CONVRT(SVB,1)
	CALL CONVRT(SVA,1)
	CALL CONVRT(N3,0)
	CALL STOSUP(1)
254	CALL CONVRT(DIST,1)
	CALL CONVRT(C,1)
	CALL CONVRT(OFF,1)
	CALL CONVRT(XAA,1)
	CALL STOSUP(2)
	END
