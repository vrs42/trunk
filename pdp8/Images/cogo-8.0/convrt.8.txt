	SUBROUTINE CONVRT(FLARG,IFLAG)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	COMMON/NNN/ADJFG
C-----WRITTEN BY COMPUTER DYNAMICS INC., BOSTON  (JULY,'68)
C JCH -- INDEX OF CHARACTER POSITION IN WORD BFR(IWD)
C IWD -- INDEX OF WORD IN ARRAY BFR
C IDTCT -- INDEX OF DATA WORD IN ARGUMENT STRING
C NMDT -- NUMBER OF ARGUMENTS FOR THE COMMAND
C IMAGE -- SHIFTED IMAGE OF WORD BFR(IWD)
	DIMENSION LBUF(15),ITEM(20),ITN(8),DGTBUF(8)
C	ALTERED DIM LIST***
C ITEM -- TABLE OF VALID ASCII CHARACTER CODES
C ITN -- TABLE OF POWERS OF TEN
	      INTEGER TGSTO,RES,BUF
	      EQUIVALENCE (BFR(1),LBUF(1))
	      DATA ITEM/'     A','     D','     +','     -',
     1'     B','     R','      ','     ,','     .',
     2'     0','     1','     2','     3','     4',
     3'     5','     6','     7','     8','     9','     /'/
	      DATA ITN/1,10,100,1000,10000,100000,1000000/
	DATA FILL/'_?`@@A'/
	DATA RSAVE/18/,ASAVE/19/,DSAVE/20/,X/-2/
	      ARG(V)= V
	      LIM= BUF(1)
C IADFL -- (USETST) SET TO -1 IF "A", 1 IF "D", AND 0 IF NEITHER
	      IADFL= 0
C IBLCT -- (BLKCT) COUNTER FOR CONSECUTIVE BLANKS IN RECORD
	      IBLCT= 0
C IPTST -- (PTTST) COUNTER FOR DIGITS FOLLOWING DECIMAL POINT
	      IPTST= 0
C JFXANS -- CONVERTED DATA IN FIXED POINT
	      JFXANS= 0
C RES -- RAW DATA PACKED IN 4 BIT BCD
	      RES= 0.
C NEGSN -- (SSIGN) SET TO -1 IF NEG SIGN, OTHERWISE 0
	      NEGSN= 0
C NODGT -- NUMBER OF DIGITS TO BE CONVERTED
	      NODGT= 0
C NEXP -- WHICH POWER OF 10 TO USE FROM 'ITN' TABLE
	NEXP=0
C---- SEARCH FOR FIRST CHARACTER= "A","D","+" OR "-"
	      KHAR= LSHIFT(IMAGE,30,0)
	      DO 6 INDEX=1,6
	IF (COMP(KHAR,ITEM(INDEX)).EQ.1) GO TO (1,2,100,4,
     . 31,32), INDEX
	    6 CONTINUE
C---- SEARCH FOR CHARACTER= BLANK,",","." OR DIGIT
	   10 KHAR= LSHIFT(IMAGE,30,0)
	      DO 9 INDEX=1,14
	IF(COMP(KHAR,ITEM(INDEX+6)).EQ.1)GO TO (5,5,7,8,8,
     .8,8,8,8,8,8,8,8,5), INDEX
	    9 CONTINUE
	      CALL ERROR(15)
	      RETURN
	31	IADFL= -3
		IDTCT= IDTCT+1
C---- FIRST CHARACTER WAS "A"
	1	IF (IADFL.EQ.0) IADFL= -2
	      CALL SLITE(2)
	      IDTCT= IDTCT+2
C---- FIRST CHARACTER WAS "D"
	    2 IADFL= IADFL+1
	32	IF (IADFL.EQ.0) IADFL= 2
C----FIRST CHARACTER WAS "+"
	      GO TO 100
C---- FIRST CHARACTER WAS "-"
	    4 NEGSN= -1
	      GO TO 100
C---- CHARACTER WAS "," OR BLANK OR "/"
	    5 IDTCT= IDTCT+1
	      IF (IDTCT.LT.NMDT) GO TO 3
	      IF (IDTCT.GT.NMDT) CALL ERROR(2)
	      KODE= 1
	      IBLCT= 0
	      IF (TGSTO.EQ.0) IBLCT= 4
	    3 IMAGE= LSHIFT(IMAGE,-6,0)
	      JCH= JCH+1
	      IF (JCH.LE.6) GO TO 111
	      JCH= 1
	      IWD= IWD+1
	      IF (IWD.LE.LIM) GO TO 11
	      IF (KODE.NE.0) GO TO 15
	IF(ADJFG.NE.0) GO TO 15
	      CALL INTRD
	      LIM= BUF(1)
	      IWD= 1
	   11 IMAGE= LBUF(IWD)
111	ZNEW=LSHIFT(IMAGE,30,0)
	IF (COMP(ZNEW,ITEM(7)).EQ.1) GO TO 12
	      IF (KODE.EQ.0).OR.(TGSTO.NE.0) GO TO 15
	      CALL ERROR(2)
	   12 IBLCT= IBLCT+1
	      IF (IBLCT.LE.6) GO TO 3
	      IF ((KODE.NE.0).AND.(TGSTO.EQ.0)) GO TO 15
	      IF (TGSTO.EQ.0) CALL ERROR(3)
	      CALL SLITE(1)
	      GO TO 15
C---- CHARACTER WAS A DECIMAL POINT
	    7 IPTST= 1
	      GO TO 100
C---- CHARACTER WAS A DIGIT
	    8 IF (IPTST.NE.0) IPTST= IPTST+1
	CALL CGET(KHAR,6,KHAR)
	KHAR=KHAR-48
	NEXP=NEXP+1
	IF(NEXP.GT.7) GO TO 22
	DGTBUF(NEXP)=KHAR
	  100 JCH= JCH+1
	      IMAGE= LSHIFT(IMAGE,-6,0)
	      IF (JCH.LE.6) GO TO 10
	      JCH= 1
	      IWD= IWD+1
	      IF (IWD.LE.LIM) GO TO 101
	      CALL INTRD
	      LIM= BUF(1)
	      IWD= 1
	  101 IMAGE= LBUF(IWD)
	      GO TO 10
C	BUILD FIXED PT #
15	NEX=1
	ADJFG=0
155	JFXANS=JFXANS+DGTBUF(NEX)*ITN(NEXP)
	NEX=NEX+1
	NEXP=NEXP-1
	IF(NEXP.GT.0)GO TO 155
	      IF (IADFL.NE.0) GO TO 20
C---- FIXED POINT VALUE TO BE STORED AS AN INTEGER
	      IF (IFLAG.GT.0) GO TO 50
	      IF (IPTST.NE.0) CALL ERROR(9)
		IF (JFXANS.GT.999) CALL ERROR(16)
	      IVALUE= ISIGN(JFXANS,NEGSN)
	   25 FLARG= ARG(IVALUE)
		IF (IADFL.EQ.-2) FLARG= ARG(0)
		IF (IADFL.EQ.-2) JD=IVALUE
	      IF (IFLAG.GE.0) GO TO 30
		IF (IVALUE.LE.0) CALL ERROR(16)
	ZNEW=XY(JFXANS,X)
	IF(COMP(ZNEW,FILL).NE.1)GO TO 30
	      WRITE(4,1000)JFXANS
	 1000 FORMAT('0PT',I5,' IS UNDEFINED')
	      IS= 4
	      CALL READ
	   20 IF (JFXANS.GT.50) GO TO 23
	      IF (IADFL.LT.0) GO TO 21
		IF (IADFL.EQ.1) FLARG= XY(JFXANS,DSAVE)
		IF (IADFL.EQ.2) FLARG= XY(JFXANS,RSAVE)
	   30 IF (NQUAD.EQ.-10) CALL READ
	      LBUF(1)= IBLCT
	      RETURN
21	XY3(5)=XY(JFXANS,ASAVE)
	      IVALUE= -999
	      GO TO 25
	   22 CALL ERROR(4)
	   23 CALL ERROR(20)
C---- FIXED VALUE TO BE STORED AS FLOATING POINT NUMBER
	   50 FLARG= ISIGN(JFXANS,NEGSN)
	      IF (IPTST.EQ.0) IPTST= 1
	      FLARG= FLARG/FLOAT(ITN(IPTST))
	      GO TO 30
	       END
