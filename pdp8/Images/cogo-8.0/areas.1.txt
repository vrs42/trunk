	SUBROUTINE AREAS(JACK)
	INTEGER BUF
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA Y/-1/,X/-2/
	      JI=JACK
	      II= BUF(3)
C     THIS STATEMENT IS NECESSARY BECAUSE OF A PRESENT BUG IN FORTRAN
	      IF(II)1,1,3
	 3    GO TO (2,156),II
	 1    I=K
		IF (II+1) 4,5,6
	4	AREA= -AREA
		GO TO 5
	6	AREA= 0.
	5      YN=0
	      XN=XY(K,Y)
	      J=K
	      BUF(3)= 1
		IF (JI.EQ.2) ACRES= 'AZIM'
		IF (JI.EQ.3) ACRES= 'BRNG'
		IF (JI.NE.1) WRITE(IP,104) ACRES
	104	FORMAT('0 PT TO PT    DIST',9X,A4/)
		IF (JI.NE.1) CALL CKLIN(2,2)
	      RETURN
	 156  YN=YN+((XY(N,Y)-XN)*(XY(J,X)-XY(K,X)))/2.0
	      J=N
	      N=K
	 160  GO TO (159,961,957,152),JI
	 159  IF(K-I)157,154,157
	 157  RETURN
	 154  K=N1
	      JI=4
	      GO TO 156
	 152  AREA= ABS(AREA+ABS(YN))
	      ACRES=AREA/43560.0
	      WRITE(IP,100) AREA,ACRES
	 100  FORMAT(6H AREA=F12.3,1X,11HSQ. FT. AND,F10.3,1X,5HACRES)
	      CALL CKLIN(1,2)
	      BUF(3)= 0
	      CALL READ
	 961  CALL DSTRAD
	      CALL RADDEG
	      WRITE(IP,101) J,N,DIST,JD,JM,B
	101   FORMAT(I4,I6,F12.3,3X,2I3,F7.3)
	      CALL CKLIN(1,2)
	      GO TO 159
	 957  CALL DSTRAD
	      CALL QUAD
	      CALL RADDEG
	      CALL ALPHA(DIRA,DIRB)
	      WRITE(IP,102) J,N,DIST,DIRA,JD,JM,B,DIRB
	102   FORMAT(I4,I6,F12.3,1X,A2,2I3,F7.3,1X,A2)
	      CALL CKLIN(1,2)
	      GO TO 159
	    2 BUF(3)= 2
	      N=K
	      N1=K
	      GO TO 160
	      END
