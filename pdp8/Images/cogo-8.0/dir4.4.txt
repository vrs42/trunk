	SUBROUTINE DIR4(KOM)
	INTEGER BUF
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA X,Y/-2,-1/
	DATA MINUS1/-1/
	KING=KOM-62
	GO TO (255,255,257,257,259,227,222,221,263,264,
     1265,266,267,268,269,200,200,200,273,273,273,
     1273,273,273,279,280,281,282),(KOM-54)
221	CALL CONVRT(NP,0)
222	CALL CONVRT(N,MINUS1)
	CALL CONVRT(N1,0)
	II=KOM-20
	IF(II.GT.20)II=47-II
	CALL PTLOC(II)
227	CALL CONVRT(N,0)
	CALL CONVRT(S,1)
	IF(KOM.EQ.27)CALL CONVRT(OFF,1)
	CALL CONVRT(L,0)
	CALL COORD(KOM-26)
255	CALL CONVRT(A,1)
	CALL CONVRT(B,1)
	ARG= 2*KOM-111
	CALL TRI(ARG)
257	CALL CONVRT(M,0)
	CALL CONVRT(J,MINUS1)
	CALL CONVRT(XL,1)
	CALL CONVRT(K,0)
	CALL CONVRT(N1,MINUS1)
	CALL CONVRT(SC,1)
	CALL CONVRT(N2,0)
	ARG= 115-2*KOM
	CALL TANG(ARG)
259	CALL CONVRT(SSC,1)
	ILIN=SSC
	GO TO 200
263	CALL CONVRT(INDEX,0)
	NMDT= NMDT+3*(INDEX-1)
 	DO 3 II=1,INDEX
	CALL CONVRT(N,0)
	CALL CONVRT(A,1)
	CALL CONVRT(B,1)
	CALL KNOW
3	CONTINUE
	GO TO 200
264	CALL CONVRT(I,MINUS1)
	CALL CONVRT(K,MINUS1)
	CALL CONVRT(J,MINUS1)
	CALL CONVRT(M,0)
	CALL DARC
	GO TO 200
271	XN= 0.
	YN= 0.
272	J= 1000
	CALL XYOUT(J,X,0)
	CALL XYOUT(J,Y,0)
	XN= -XN
	YN= -YN
265	CALL CONVRT(I,0)
	CALL CONVRT(K,0)
	IF (KING.GE.19) GO TO 1
	CALL CONVRT(J,MINUS1)
282	CALL CONVRT(YN,1)
	CALL CONVRT(XN,1)
	JD= 0
	JM= 0
	SC=0.
	IF (KING.NE.3) GO TO 272
281	CALL GETANG(1)
10	IF (KING.EQ.19) GO TO 271
1	CALL CORCON
200	CALL READ
266	CALL CONVRT(R,1)
	CALL CONVRT(T,1)
	CALL CONVRT(XL,1)
	CALL CONVRT(YN,1)
277	CALL GETANG(1)
	GO TO 4
267	CONTINUE
268	CALL GETANG(1)
8	CALL CONVRT(N1,0)
4	CALL CURVE(KING)
	GO TO 200
269	CALL GETANG(2)
	GO TO 268
280	N4= 1
279	CALL CONVRT(J,MINUS1)
	CALL CONVRT(N,MINUS1)
273	CALL CONVRT(N1,0)
	CALL SAVE(KING)
	GO TO 200
	END
