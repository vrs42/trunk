	SUBROUTINE INTA
	INTEGER TGSTO,BUF,XYPOS
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	COMMON/KKK/KOM,COMND
	COMMON/PPP/LIM
	DIMENSION SPEC(6,4),COM0(118),LBUF(15),COM1(41),COM2(39)
	DIMENSION COM3(38)
	EQUIVALENCE (BFR(1),LBUF(1)),(COM0,COM1(1)),(COM0(42),COM2(1))
	EQUIVALENCE (COM0(81),COM3(1))
	DATA MASK1/'????@@'/,MASK2/'@?????'/,MASK3/'?@@@@@'/
	DATA SPEC/'ADF   ','ACD   ','ATD   ','ADJC  ','HEAD  ','STAR  ',
     1'AAN   ','ACA   ','ATA   ','ADJB  ','DUMY  ','FINI  ',
     2'AAZ   ','ACZ   ','ATZ   ','ADJU  ','RETU  ','NOLI  ',
     3'ABR   ','ACB   ','ATB   ','ADJA  ','READ  ','LIST  '/

	DATA COM1/'   CLR','   DMP','   STR','   LAZ','   IAZ','   LBR',
     1'   IBR','   LAN','   LLN','   LDF','   ANG','   ASV','   PLN',
     1'   DST','   DSV','   RDF','     S','   SPL','   SMI','   DCV',
     1'   OAL','   SPA','   DLN','   DSL','   GRD','   ALN','   COF',
     1'   CPA','   PIN','   AIN','   BIN','   AAI','   ALP','   ALA',
     1'   SLI','   SCI','   SSI','    CS','777777','777777','    AR'/
	DATA COM2/'   ARA','   ARB','    ES','    SS','    OF','   SSP',
     1'    SE','   ORG','    PI','   TRM','   SUB','   SUP','   SEG',
     1'   RTL','   RTH','    ET','    CT','    LN','   CEP','   SEP',
     1'   OEA','   KWN','   DAE','   CCN','   SCV','   RAA','   RAC',
     1'   AAG','   TTS','   TTG','   INP','   SVA','   SVC','   SVR',
     1'   SVT','   ISV','   RSV','   ZSV','   BSV'/
	DATA	COM3/'   ROT','   TAX','   PER',
     2'   LAT','   DEP','   SLN','   SAR','   SDS','   SBR','   CAN',
     3'   CEN','   ARP','   ARM','   ALB','   AAC','   APC','   AZC',
     4'   ABC','   TRP','   SMP','   FSC','   TNG','   TOF','   ZOF',
     5'   BOF','   SUR','   ROF','   DSP','   SOF','   CSP','   OUT',
     6'   PPS','   PPG','   SYS','   SYR','777777','777777','777777'/
	IF(NEWVAR.EQ.1)GO TO 999
	IX=0
	IF(N6.GE.0) GO TO 1020
	N6=IABS(N6)
	KOM=N6
C-------BUF(1) SET UP IN READ
1020	LIM=BUF(1)
	IWD=1
	ZNEW=LSHIFT(LBUF(1),30,0)
C-------IS IT A COMMENT?
	IF(COMP(ZNEW,'     *').EQ.1)CALL READ
C-------IS IT A CONTINUE CMD RO A BLANK CMD FIELD?
	IF(COMP(ZNEW,'     "').EQ.1)GO TO 113
	ZNEW=LSHIFT(LBUF(1),18,0)
	IF(COMP(ZNEW,'      ').EQ.1)GO TO 113
C-------CHECK FOR A SPECIAL CMD
	JCH=LSHIFT(LBUF(1),0,MASK1)
	DO 1 JJ=1,4
	DO 1 IJ=1,6
	IF(COMP(SPEC(IJ,JJ),JCH).EQ.1)GO TO 10
1	CONTINUE
20	TGSTO=0
	DO 2 IWD=IWD,LIM
	RES='      '
	IMAGE=LBUF(IWD)
	DO 3 JCH=1,6
	KHAR=LSHIFT(IMAGE,30,0)
	IF(COMP(KHAR,'      ').EQ.1)GO TO 7
	RES=LSHIFT(RES,-6,0)
	RES=LOR(RES,KHAR)
3	IMAGE=LSHIFT(IMAGE,-6,0)
2	CONTINUE
7	DO 12 IX=1,118
	IF(COMP(RES,COM0(IX)).EQ.1)GO TO 30
12	CONTINUE
65	CALL ERROR(1)
C-------IT IS A SPECIAL CMD
10	IX=6*(JJ-1)+IJ
	COMND=SPEC(IX,1)
	IF(IJ.LE.3)JJ=JJ-3
	IF(JJ.LT.0)JJ=JJ+1
150	GO TO (105,105,105,102,120,107,105,105,105,103,65,109,106,106
     9,106,101,108,110,106,106,106,104,116,111),IX
100	IX=6*(KK-1)+4
	GO TO 150
C-------IT IS A CONTINUE CMD(SPACES OR QUOTES)
113	DO 5 KK=1,4
	IF(COMP(SPEC(4,KK),COMND).EQ.1)GO TO 100
5	CONTINUE
C-------INTB SETS UP NMDT
	CALL INTB(KOM)
47	IMAGE=LSHIFT(LBUF(1),-6,0)
	JCH=2
	GO TO 33
C**MOST OF THE SPECIAL COMMAND CODE FOLLOWS.
105	COMND='ADJB  '
	GO TO 1115
106	COMND='ADJA  '
1115	KOM=-1
	N3=IJ
	N4=JJ
	NMDT=2
	JCH=4
	IMAGE=LSHIFT(LBUF(IWD),-18,0)
	GO TO 33
C-------MASK OUT FIRST CHAR
120	LBUF(2)=LSHIFT(LBUF(2),0,MASK2)
107	CALL START
110	IF(IPLOT.LE.0)IPLOT=IPLOT+1
	GO TO 201
111	IF(IPLOT.GE.0)IPLOT=IPLOT-1
	GO TO 201
109	CALL CKLIN(-1,3)
C-------WRITE CURRENT BLK OF VIRTUAL ARRAY
C------READ LAST BLK OF VIRTUAL ARRAY BEFORE EXITING
	WRITE(2'XYPOS) ARRAYX
	READ(2'39) ARRAYX
	CALL EXIT
102	COMND=COM0(40)
	KOM=40
	NMDT=5
	GO TO 47
103	COMND='ADJC  '
1114	KOM=39
	NMDT=6
	GO TO 47
104	COMND=COM0(39)
	GO TO 1114
116	IQ=4
	GO TO 200
C-------ADJU IS ILLEGAL WITH SHORT FORMS
101	CALL ERROR(1)
30	COMND=COM0(IX)
	TGSTO=0
	IF(IX.EQ.83).OR.(IX.EQ.92).OR.(IX.EQ.93)TGSTO=1
	IF(IX.GE.41).AND.(IX.LE.43)TGSTO=1
	CALL INTB(IX)
C-------LOOK FOR BEGINNING OF DATA ARGS
	KOM=IX
33	IX=IWD
	IF(IWD.LE.LIM)GO TO 36
	CALL SLITE(1)
	GO TO 46
C-------LOOK FOR 1ST NON-SPACE CHAR AFTER CMD
36	DO 40 IWD=IX,LIM
	J1=1
	IF(IWD.EQ.IX)J1=JCH
	IF(IWD.NE.IX)IMAGE=LBUF(IWD)
	DO 40 JCH=J1,6
C-------IS 1ST CHAR A SPACE?
	ZNEW=LSHIFT(IMAGE,0,MASK3)
	IF(COMP(ZNEW,'      ').NE.1)GO TO 45
40	IMAGE=LSHIFT(IMAGE,-6,0)
	IF(LBUF(3).EQ.0)CALL ERROR(7)
	CALL SLITE(1)
	GO TO 46
45	KODE=0
	IDTCT=0
	IF((KOM.LE.72).AND.(KOM.GE.70))GO TO 115
	IF((KOM.LE.113).AND.(KOM.GE.111))GO TO 115
	IF((KOM.EQ.114).OR.(KOM.EQ.115)) GO TO 180
	CALL DIRECT(KOM)
C**RETURN FROM AREA SUBS. NMDT VARIES
999	NEWVAR=0
	IF(BUF(3).EQ.0)CALL READ
46	CALL SLITET(1,L1)
	IF(L1.EQ.2)GO TO 33
	CALL INTRD
	IWD=1
	LIM=BUF(1)
	IMAGE=LBUF(1)
	JCH=1
	GO TO 33
C-------GET LOGICAL UNIT FOR ACCESSING M.S. FILE
115	CALL CONVRT(IQ,0)
	IF((IQ.EQ.1).OR.(IQ.EQ.3).OR.(IQ.EQ.4)) GO TO 181
	GO TO 182
181	IF(KOM.EQ.72) GO TO 200
	IF(KOM.EQ.111) GO TO 160
182	IF((IQ.GT.9).OR.(IQ.LT.5))CALL ERROR(999)
	IF((KOM.EQ.71).OR.(KOM.EQ.70)) GO TO 114
	IF((KOM.EQ.112).OR.(KOM.EQ.113)) GO TO 112
	IF (KOM.EQ.111) GO TO 160
	IO=IQ
C-------COME HERE WHEN DOING RETURN/FILE
108	IF(IO.LT.5)CALL ERROR(8)
	IS=IO
	CALL READ
160	IP=IQ
	CALL READ
200	IS=IQ
201	CALL READ
114	CALL CONVRT(N,0)
	CALL CONVRT(J,0)
	I=71-KOM
	CALL TABLES(I,IQ,N,J)
112	CALL CONVRT(N,0)
	CALL CONVRT(J,0)
	I=114-KOM
	CALL POINTS(I,IQ,N,J)
180	CALL CONVRT(IQ,0)
	IQ=115-KOM
	CALL SYSMK(IQ)
	END
