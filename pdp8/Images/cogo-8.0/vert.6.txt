	SUBROUTINE VERT(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA SVPI,ZVPI,NVPI,XA,XB/7,8,9,10,11/
	      GO TO (1,2,3),II
C     ORIGIN AND TERMINUS READ N,S,E1
	 1    DIFF=0
	CALL XYOUT(1,XB,1.)
C     SET FLAG TO SHOW COMMAND PROCESSED
	      I=1
4	CALL XYOUT(I,NVPI,N)
	CALL XYOUT(I,ZVPI,E1)
	CALL XYOUT(I,SVPI,S)
	      GO TO (5,5,6),II
	 5    CALL READ
C     PI READ,N,N1,N2,S,E1,E2,E3
	 2    I=I+1
	      IF (I.GT.29) GO TO 15
	CALL XYOUT(I,XB,E2)
	CALL XYOUT(I,XA,E3)
	      GO TO 4
	   15 CALL SLITET(1,L1)
	      IF (L1.EQ.1) GO TO 16
	      WRITE(4,500)
	  500 FORMAT(28H MORE THAN 28 PI'S INDICATED)
	      CALL CKLIN(1,1)
	      I= 29
	CALL XYOUT(30,NVPI,N)
	   16 CALL SLITE(1)
	      CALL READ
	 3    I=I+1
	      CALL SLITET(1,L1)
	XX(1)=XY(30,NVPI)
	      IF ((L1.EQ.2).OR.(XX(1).EQ.N)) GO TO 4
	      WRITE(4,501) XX(1)
	  501 FORMAT(' TERMINUS SHOULD BE SET AT PT',I5)
	      CALL CKLIN(1,1)
	      IS= 4
	      GO TO 16
	 6    NTOT=I
	      NSAVE=NTOT-1
	      NQ=NTOT-1
	      DO 7 NN=2,NQ
	      N=NN
	      CALL PCPT
	      GBP=100.0*GB
	      GAP=100.0*GA
	      GMP=100.0*GM
	XX(2)=XY(NN,NVPI)
	      WRITE(IP,20) XX(2)
	 20   FORMAT(21H CURVE WITH PI NUMBER I6)
	      WRITE(IP,8) SVPC,ZVPC,GBP
	8     FORMAT(' PC STA=',F10.3,', EL=',F9.3,
     1', GRADE AHEAD=',F7.3)
	YY(1)=XY(NN,SVPI)
	YY(2)=XY(NN,ZVPI)
	      WRITE(IP,9) YY(1),YY(2),B,GMP
9     FORMAT(' PI STA=',F10.3,', EL=',F9.3,', EL ON CURVE=',F9.3/
     1' GRADE ON CURVE AT PI=',F7.3)
		IF(GBP*GAP.GT.0.) GO TO 21
		HILO=5H HIGH
		IF(GAP.GT.GBP) HILO=5H  LOW
	GP=ABS(GB-GA)/(XY(NN,XA)+XY(NN,XB))
		B=ABS(GB)/GP
		GP=.5*GP*B*B
		GP=ZVPC+B*GB-SIGN(GP,GB)
		B=SVPC+B
		WRITE(IP,30) HILO,B,GP
	30	FORMAT(' ',A5,' PT AT STA=',F10.3,', EL=',F9.3)
	CALL CKLIN(1,2)
	21	WRITE(IP,10) SVPT,ZVPT,GAP
	10	FORMAT(' PT STA=',F10.3,', EL=',F9.3,
     1', GRADE AHEAD=',F7.3/)
	      CALL CKLIN(6,2)
	 7    CONTINUE
	      CALL READ
	      END
