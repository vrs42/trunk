C	CURVE.1
	SUBROUTINE CURVE(II)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	COMMON/BBB/JCH,IWD,IDTCT,NMDT,IMAGE,TGSTO,BFR(15),KODE,IO,IS
	COMMON/CCC/N,N1,N2,N3,N4,N5,NP,NB,NA,I,J,K,K2,L,M,R
	COMMON/DDD/BA,APC,SIGGN,AA,SPT
	COMMON/EEE/APT,AREA,ACRES,JD,JM,SC,JJD,JJM,SSC,NQUAD,NNQUAD
	COMMON/FFF/A,B,C,DIST,YD,YM,YS,XN,YN,T,SB
	COMMON/GGG/XL,S,OFF,BUF(13),DX,DY,XAA,YAA,IPG,ILIN,E1,E2
	COMMON/HHH/E3,E4,E5,XX(3),YY(3),HOLL(10),ID,IA
	COMMON/III/IPRES,DELTA,N6,HBA,HAA,NSAVE,DIFF,SVPT,SVPC
	COMMON/JJJ/ZVPC,ZVPT,NTOT,GB,GA,AB,BB,SVB,SVA
	COMMON/LLL/GM,IPLOT,IP
	DATA ASAVE,DSAVE/19,20/
		TAN(U)= SIN(U)/COS(U)
		II= II-3
		CALL ANGRAD
		R= ABS(R)
		IF (II.GE.25) GO TO 1
		GO TO (1,2,3,4), II
	4	YD= A
		JD= JJD
		JM= JJM
		SC= SSC
		CALL ANGRAD
		A= A+YD
		IF (A.LT.0.) A= 6.2831853+A
		CALL RADDEG
		WRITE(IP,40) N1,JD,JM,B
	40	FORMAT(' ANGLE(',I3,')=',I4,I3,F7.3)
		CALL CKLIN(1,2)
	IF (N1.NE.0) CALL XYOUT(N1,ASAVE,A)
		GO TO 100
	3	R= 50./SIN(A/2.)
		GO TO 5
	2	R=100./A
	5	IF (ABS(A).LE..001) R= 99999.999
		WRITE(IP,50) N1,R
	50	FORMAT(' RADIAL DIST(',I3,')=',F10.3)
		CALL CKLIN(1,2)
	IF (N1.NE.0) CALL XYOUT(N1,DSAVE,R)
		GO TO 100
	1	IF (R.LE.0.) GO TO 10
		IF (T.LE.0.) GO TO 6
		A= ATAN(T/R)*2.
		YN= R*SIN(A/2.)*2.
		GO TO 16
	6	IF (XL.LE.0.) GO TO 8
		A= XL/R
	7	T=R*TAN(A/2.)
	15	YN= R*SIN(A/2.)*2.
		GO TO 90
	8	IF (YN.LE.0.) GO TO 9
		A= ASIN(YN/R/2.)*2.
		T=R*TAN(A/2.)
		GO TO 16
	9	IF (A.LE.0.) GO TO 99
		XL= R*A
		GO TO 7
	10	IF (T.LE.0.) GO TO 20
		IF (XL.LE.0.) GO TO 11
		A= 0.0000001
		DA= 0.0174533
		Y2= XL/A*TAN(A/2.)-T
	17	Y1= Y2
		A= A+DA
		Y2= XL/A*TAN(A/2.)-T
		IF ((A.GT.6.2831853).OR.(ABS(DA).LT.0.0000001)) GO TO 99
		IF (Y2*Y1) 18,14,17
	18	IF (ABS(Y2).LE.0.0001) GO TO 14
		DA= -DA/2.
		GO TO 17
	14	R= XL/A
		GO TO 15
	11	IF (YN.LE.0.) GO TO 12
		A= ACOS(YN/T/2.)*2.
	13	R= T/TAN(A/2.)
		 GO TO 16
	12	IF (A.LE.0.) GO TO 99
		YN=T*COS(A/2.)*2.
		GO TO 13
	20	IF (XL.LE.0.) GO TO 30
		IF (YN.LE.0.) GO TO 21
		A= 0.0000001
		DA= 0.0174533
		Y2= XL/A*2.*SIN(A/2.)-YN
	19	Y1= Y2
		A= A+DA
		Y2= XL/A*2.*SIN(A/2.)-YN
		IF ((A.GT.6.2831853).OR.(ABS(DA).LT.0.0000001)) GO TO 99
		IF (Y2*Y1) 22,23,19
	22	IF (ABS(Y2).LE.0.0001) GO TO 23	
		DA= -DA/2.
		GO TO 19
	23	R=XL/A
		T= YN/COS(A/2.)/2.
		GO TO 90
	21	IF (A.LE.0.) GO TO 99
		R= XL/A
		GO TO 7
	30	IF (YN.LE.0.) GO TO 99
		R= YN/SIN(A/2.)/2.
		T= YN/COS(A/2.)/2.
	16	XL= R*A
	90	CALL RADDEG
		WRITE(IP,60) R,T,XL,YN,JD,JM,B
60	FORMAT(4X,'RADIUS   TANGENT     ARC',6X,'CHORD',6X,'DELTA'/
     1 F11.3,3F10.3,I4,I3,F7.3)
		CALL CKLIN(2,2)
		XN= A
		IF ((II.EQ.25).OR.(II.EQ.26)) CALL INVAZ
		IF (II.NE.26) GO TO 100
		SIGGN= ISIGN(1,N1)
		B= 1.5707963-XN/2.
		CALL DSTRAD
		A= A+SIGGN*B
		CALL RADDEG
		SC= B
		XXN1=N1
		DIST=SIGN(R,XXN1)
		N= IABS(N1)
		N1= N
		CALL LOCAZ
	100	CALL READ
	99	WRITE(4,70)
	70	FORMAT(' NO SOLUTION POSSIBLE WITH THESE DATA')
		CALL CKLIN(1,1)
		GO TO 100
		END
