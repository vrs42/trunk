	SUBROUTINE SYSMK(IQ)
	COMMON/AAA1/XY3(6),IMX,IPX,ARRAYX(85)
	COMMON/AAA/SLTVAR,NEWVAR,XYPOS,SAPOS
	DIMENSION DUM(1)
	EQUIVALENCE(DUM(1),SLTVAR)
	IF(IQ.EQ.0) GO TO 5
	WRITE(2'XYPOS) ARRAYX
	DO 1 I=1,12
1	ARRAYX(I)=DUM(I)
	DO 2 I=1,73
2	ARRAYX(I+12)=DUM(I+97)
	WRITE(2'38)ARRAYX
	DO 3 I=1,75
3	ARRAYX(I)=DUM(I+170)
	WRITE(2'39)ARRAYX
	CALL EXIT

5	READ(2'38)ARRAYX
	DO 6 I=1,12
6	DUM(I)=ARRAYX(I)
	DO 7 I=1,73
7	DUM(I+97)=ARRAYX(I+12)
	READ(2'39)ARRAYX
	DO 8 I=1,75
8	DUM(I+170)=ARRAYX(I)
	READ(2'1)ARRAYX
	XYPOS=1
	CALL READ
	END
