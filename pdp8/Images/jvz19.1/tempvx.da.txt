        SUBROUTINE LSQSE(BX,SBX,AI,SEA,BS,SEB,NSTAR,NSTOP,X,Y,COR)
C       LSQSTL DOES STD ERRORS OF BS AND AI, ALSO BX.
C       BX,SBX ARE SLOPE AND STD ERROR FOR ZERO INTERCEPT
C       AI,SEA;BS,SEB ARE APRAMETERS AND ERRORS WITH INTERCEPT.
C
        DIMENSION X(50),Y(50)
C*****  DOUBLE PRECISION SUMX,SUMY,SUMXQ,SUMXY,SUMYQ,FN
C*****  DOUBLE PRECISION SXX,SXY,SYY,SQ,FN2,FN1
C       STRAIGHT LINE LEAST SQUARES
        N=NSTOP-NSTAR+1
        FN=N
        SUMX=0.
        SUMY=0.
        SUMXQ=0.
        SUMXY=0.
        SUMYQ=0.
        J=NSTAR-1
        DO 10 I=1,N
        J=J+1
        XJ=X(J)
        YJ=Y(J)
        SUMX=SUMX+XJ
        SUMY=SUMY+YJ
        SUMXQ=SUMXQ+XJ**2
        SUMYQ=SUMYQ+YJ**2
10      SUMXY=SUMXY+XJ*YJ
C** LINE WITH INTERCEPT:
        SXY=SUMXY-SUMX*SUMY/FN
        SXX=SUMXQ-SUMX*SUMX/FN
        BS=SXY/SXX
        AI=SUMY/FN-BS*SUMX/FN
        SYY=SUMYQ-SUMY*SUMY/FN
        FN2=N-2
        SEB=SQRT(ABS((SYY-SXY**2/SXX))/(FN2*SXX))
        SEA=SEB*SQRT(SUMXQ/FN)
C** LINE THRU ZERO:
        FN1=N-1
        BX=SUMXY/SUMXQ
        SQ=ABS((SUMYQ-BX*SUMXY)/FN1)
        SBX=SQRT(SQ/SUMXQ)
C** CORRELATION COEFF:
        COR=SXY/SQRT(ABS(SXX*SYY))
        RETURN
        END
