C  MOR3.F4   TWO SPHERES BY METHOD OF REFLECTIONS - CDC VERSION
C       PERPENDICULAR
C
        PROGRAM MOR2(INPUT,OUTPUT,TAPE4=INPUT,TAPE5=OUTPUT)
        DIMENSION SUMB(160,9,9),SUMC(160,9,9),SUMD(160,9,9)
        DIMENSION B(160,9,9),C(160,9,9),D(160,9,9)
C
        WRITE(5,1500)
1500    FORMAT(51H TWO SPHERES BY THE METHOD OF REFLECTIONS - PERPEND)
C
        N=9
C
        NMAX=N*N-1
        NPOW=N
        IFLECT=N+1
C
        NQ=NMAX*2
        DO 100 I=1,NQ
        DO 100 J=1,NPOW
        DO 100 K=1,NPOW
        B(I,J,K)=0.
        C(I,J,K)=0.
        D(I,J,K)=0.
        SUMB(I,J,K)=0.
        SUMC(I,J,K)=0.
100     SUMD(I,J,K)=0.
        B(2,1,1)=.25
        C(2,1,1)=1.5
        SUMB(2,1,1)=.25*2./3.
        SUMC(2,1,1)=1.5*2./3.
C
        II=1
        DO 500 IR=2,IFLECT
        JJ=II
        II=3-II
C
        DO 300 NM1=1,NMAX
        NI1=NM1+(II-1)*NMAX
        NJ1=NM1+(JJ-1)*NMAX
        Q=NM1
        N1=SQRT(Q)
        M1=NM1-N1*N1-N1
C
        DO 110 I=1,NPOW
        DO 110 J=1,NPOW
        B(NI1,I,J)=0.
        C(NI1,I,J)=0.
110     D(NI1,I,J)=0.
C
        BB=N1*(2.*N1-1.)/(2.*(N1+1.))
        BNC=1./(2.*(N1+1.))
        BC=N1*(2.*N1+1.)/(2.*(2.*N1+3.)*2.*(N1+1.))
        BD=BB/N1
        CB=N1*(2.*N1-1.)*(2.*N1+1.)/(N1+1.)
        CNC=(2.*N1+1.)/(N1+1.)
        CC=N1*(2.*N1-1.)/(2.*(N1+1.))
        CD=CB/N1
        DD=(1.)/(N1+1.)
        DC=DD/N1
C
        DO 200 NM2=1,NMAX
        NI2=NM2+(II-1)*NMAX
        NJ2=NM2+(JJ-1)*NMAX
        Q=NM2
        N2=SQRT(Q)
        M2=NM2-N2*N2-N2
C
        F1=FUNC2(N1,M1,N2,M2,II)
        F2=FUNC2(N1,M1+1,N2,M2,II)
        DEL=FUNC3(N1,M1,N2,M2)
        SIG=FUNC4(N1,M1,N2,M2)
        AN=N1*(2.-N2)*(2.*N1-1.)+DEL*(N2*N1+1.-2.*N1-2.*N2)
        AN=AN/(2.*N2*(2.*N2-1.))
C
        DO 150 I=1,NPOW
        DO 150 J=1,NPOW
        I1=J+N1-1
        IF(I1.GT.NPOW)GOTO 150
        I2=I+N2
        IF(I2.GT.NPOW)GOTO 150
        B(NI1,I1,I2)=B(NI1,I1,I2)+F1*BNC*AN*C(NJ2,I,J)
        C(NI1,I1,I2)=C(NI1,I1,I2)+F1*CNC*AN*C(NJ2,I,J)
        I2=I2+1
        IF(I2.GT.NPOW)GOTO 120
        B(NI1,I1,I2)=B(NI1,I1,I2)+F2*SIG*BD*D(NJ2,I,J)
        C(NI1,I1,I2)=C(NI1,I1,I2)+F2*SIG*CD*D(NJ2,I,J)
        I2=I2+1
        IF(I2.GT.NPOW)GOTO 120
        B(NI1,I1,I2)=B(NI1,I1,I2)+F1*BB*B(NJ2,I,J)
        C(NI1,I1,I2)=C(NI1,I1,I2)+F1*CB*B(NJ2,I,J)
120     I1=I1+1
        I2=I2-2
        IF(I1.GT.NPOW)GOTO 150
        D(NI1,I1,I2)=D(NI1,I1,I2)-F2*SIG/N2*DC*C(NJ2,I,J)
        I1=I1+1
        IF(I1.GT.NPOW)GOTO 130
        B(NI1,I1,I2)=B(NI1,I1,I2)+F1*BC*C(NJ2,I,J)
        C(NI1,I1,I2)=C(NI1,I1,I2)+F1*CC*C(NJ2,I,J)
130     I2=I2+1
        I1=I1-1
        IF(I2.GT.NPOW)GOTO 150
        D(NI1,I1,I2)=D(NI1,I1,I2)-F1*N2*DD*D(NJ2,I,J)
150     CONTINUE
C
200     CONTINUE
        DO 250 I=1,NPOW
        DO 250 J=1,NPOW
        TWO=4.**(I+J-2)*2./3.
C       TWO=1.
        SUMB(NI1,I,J)=SUMB(NI1,I,J)+B(NI1,I,J)*TWO
        SUMC(NI1,I,J)=SUMC(NI1,I,J)+C(NI1,I,J)*TWO
250     SUMD(NI1,I,J)=SUMD(NI1,I,J)+D(NI1,I,J)*TWO
300     CONTINUE
500     CONTINUE
        DO 600 NM1=1,3
        NI1=NM1+NMAX
        Q=NM1
        N1=SQRT(Q)
        M1=NM1-N1*N1-N1
C
        IF(M1.NE.0)GOTO 501
        WRITE(5,1010)N1,M1
        DO 510 I=1,NPOW
        WRITE(5,1000)(SUMB(NM1,I,J),J=1,NPOW)
510     CONTINUE
        WRITE(5,1020)N1,M1
        DO 520 I=1,NPOW
        WRITE(5,1000)(SUMB(NI1,I,J),J=1,NPOW)
520     CONTINUE
        WRITE(5,1030)N1,M1
        DO 530 I=1,NPOW
        WRITE(5,1000)(SUMC(NM1,I,J),J=1,NPOW)
530     CONTINUE
        WRITE(5,1040)N1,M1
        DO 540 I=1,NPOW
        WRITE(5,1000)(SUMC(NI1,I,J),J=1,NPOW)
540     CONTINUE
C
501     IF(M1.EQ.0)GOTO 600
        WRITE(5,1050)N1,M1
        DO 550 I=1,NPOW
        WRITE(5,1000)(SUMD(NM1,I,J),J=1,NPOW)
550     CONTINUE
        WRITE(5,1060)N1,M1
        DO 560 I=1,NPOW
        WRITE(5,1000)(SUMD(NI1,I,J),J=1,NPOW)
560     CONTINUE
600     CONTINUE
1000    FORMAT(5G25.18)
1010    FORMAT(13H B - UA   N= ,I3,5H  M= ,I3)
1020    FORMAT(13H B - UB   N= ,I3,5H  M= ,I3)
1030    FORMAT(13H C - UA   N= ,I3,5H  M= ,I3)
1040    FORMAT(13H C - UB   N= ,I3,5H  M= ,I3)
1050    FORMAT(13H D - UA   N= ,I3,5H  M= ,I3)
1060    FORMAT(13H D - UB   N= ,I3,5H  M= ,I3)
        STOP
        END
C  FUNC2.F4    (CALLED BY MOR2.F4)
        FUNCTION FUNC2(N1,M1,N2,M2,II)
        FUNC2=0.
C
        LSUM=N1+N2+M1-M2
        IF (LSUM.NE.LSUM/2*2)GOTO 900
C
        IF(II.NE.2)GOTO 100
        IF(M2.GT.N2)GOTO 900
        ID=IABS(M2-M1)
        IF(ID.GT.N1+N2)GOTO 900
C
100     LSUM=LSUM/2
C
        FUNC2=1.
        DO 200 I=1,LSUM
200     FUNC2=FUNC2*(LSUM+I)
        LSUM=(N1+N2+M2-M1)/2
        DO 300 I=1,LSUM
300     FUNC2=FUNC2*(LSUM+I)
        NQ=N2-M2
        DO 400 I=1,NQ
400     FUNC2=FUNC2/I
        NQ=N1+M1
        DO 500 I=1,NQ
500     FUNC2=FUNC2/I
        FUNC2=FUNC2*(-1)**LSUM/2.**(N1+N2)
900     RETURN
        END
C  FUNC3.F4      (CALLED BY MOR2.F4)
        FUNCTION FUNC3(N1,M1,N2,M2)
        FUNC3=0.
        IF(N1+N2+M1-M2-1.EQ.0)RETURN
        IF(N1+N2+M2-M1-1.EQ.0)RETURN
        FUNC3=(N1+M1)*(N1+M1-1.)/(N1+M1+N2-M2-1.)
        FUNC3=FUNC3+(N1-M1)*(N1-M1-1.)/(N1-M1+N2+M2-1.)
        RETURN
        END
C  FUNC4.F4   (CALLED BY MOR2.F4)
        FUNCTION FUNC4(N1,M1,N2,M2)
        FUNC4=0.
        IF(N1+M1+N2-M2.EQ.0)RETURN
        FUNC4=(M2*N1+N2*M1)*(N1+M1+1.)/(N1+M1+N2-M2)
        RETURN
        END
       IF(N1+N2+M2-M1-1.EQ.0)RETURN
        FUNC3=(N1+M1)*(N1+M1-1.)/(N1+M1+N2-M2-1.)
        FUNC3=FUNC3+(N1-M1)*(N1-M1-1.)/(N1-M1+N2+M2-1.)
        RETURN
        END
C  FUNC4.F4   (CALLED BY MOR2.F4)
