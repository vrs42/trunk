C  SECOM4.F4
C  DIAGONAL COMPARISON PROGRAM - INTERNAL COMPARISON
C  TO GET ALGORITHM FOR DIAGONAL SEARCH LIKE SECOM5.FOR SERIES ON
C  VAX.
C  UNITS:
C       UNIT 1 IS SCORING SCHEME
C       UNIT 2 IS SEQUENCE 1
C
C  NUMBER OF DIAGONALS SHOULD BE (L1-LD+1)*(L2-LD+1)
C
        DIMENSION ITAB(27,27),ISEQ1(500),ISEQ2(500)
        DIMENSION ICHAR(27),TITLE(10),ISEQ(500),LIST(3,101)
        DIMENSION TIT(2)
        DOUBLE PRECISION AVE,SIG,COUNT,VN,DF,DSQRT
        LOGICAL NSW
        DATA ICHAR/'A','B','C','D','E','F','G','H','I','J','K','L','M',
     1'N','O','P','Q','R','S','T','U','V','W','X','Y','Z',' '/
C
        WRITE(4,999)
999     FORMAT(' SECOM4.F4 - INTERNAL HOMOLOGY'/'      SSW 0 = 1 STOPS P
     1RINT LIST')
C  READ TABLE ON UNIT 1
        LMAX=500
        READ(1,1000)TITLE
        WRITE(4,1000)TITLE
1000    FORMAT(1X,10A6)
        DO 5 J=1,27
5       READ(1,1005)(ITAB(I,J),I=1,27)
1005    FORMAT(27I2)
C  READ FIRST SEQUENCE:
        READ(2,1009)NS1,NST1,NSTP,TIT
1009    FORMAT(3I5,2A6)
        WRITE(4,1009)NS1,NST1,NSTP,TIT
        IF(NST1.EQ.0)NST1=1
1010    FORMAT(I5,10A6)
        READ(2,1010)NRES,TITLE
        WRITE(4,1010)NRES,TITLE
C*****DISABLE:
        READ(2,1011)(ISEQ(J),J=1,NRES)
        WRITE(4,1012)(ISEQ(J),J=1,NRES)
        DO 12 I=1,NRES
C  CHECK END OF SEQUENCE:
        IF(ISEQ(I).NE.'>')GO TO 13
        NRES=I
        GO TO 15
13      DO 11 J=1,27
        IF(ISEQ(I).NE.ICHAR(J))GO TO 11
        ISEQ1(I)=J
        GO TO 12
11      CONTINUE
        ISEQ1(I)=27
12      CONTINUE
1011    FORMAT(30A1)
1012    FORMAT(30(1X,A1))
15      L1=NRES-1
C  SECOND SEQUENCE:
        L2=L1
        DO 20 I=1,L2
20      ISEQ2(I)=ISEQ1(I)
        WRITE(4,1025)L1
1025    FORMAT(' LENGTH L1 ='I4)
26      DO 27 I=1,101
        DO 27 K=1,3
27      LIST(K,I)=0
        LIST(1,101)=4000
        WRITE(4,1030)
1030    FORMAT('0 DIAGONAL LENGTH (99=END):'$)
        READ(4,1031)LD
1031    FORMAT(I5)
        IF(LD.EQ.99)STOP
        WRITE(4,1032)
1032    FORMAT(' TARGET SCORE:'$)
        READ(4,1033)TS
1033    FORMAT(F9.0)
        IF(TS.EQ.99.)STOP
        AVE=0.
        SIG=0.
        COUNT=0.
        BIGR=0.
        IR=1
        MAXL=L1
        IF(L2.GT.MAXL)MAXL=L2
35      KL2=2
        KL3=3
C  JC IS LOOPED TO 125
C  JC=2 MEANS DIAG 0 IS NOT EXAMINED.  IN SECOM5.F4, JC=IJK
        JC=2
C  SET UP 0 DIAG 1ST TIME:
40      NV=0
        DO 50 I=1,LD
        IS=ISEQ2(IR+I-1)
        JS=ISEQ1(JC+I-1)
        NV=NV+ITAB(IS,JS)
50      CONTINUE
C  NR, # RES ALONG A DIAGONAL; NT IS # SLIDES
        NJ=L1-JC+1
        NI=L2-IR+1
        NR=NJ
        IF(NI.LT.NJ)NR=NI
        NT=NR-LD
        J=0
        GO TO 70
60      J=J+1
        IF(J.GT.NT)GO TO 125
        IS=ISEQ2(IR+J-1)
        JS=ISEQ1(JC+J-1)
        NV=NV-ITAB(IS,JS)
        IS=ISEQ2(IR+LD+J-1)
        JS=ISEQ1(JC+LD+J-1)
        NV=NV+ITAB(IS,JS)
C  AVERAGE AND COUNT:
70      VN=FLOAT(NV)
        AVE=AVE+VN
        SIG=SIG+VN*VN
        COUNT=COUNT+1.D0
        IF(VN.GE.TS)BIGR=BIGR+1.
        K=2
80      KM1=K-1
        IF(NV.LE.LIST(1,K))GO TO 90
        DO 85 IK=1,3
85      LIST(IK,KM1)=LIST(IK,K)
        K=K+1
        GO TO 80
90      IF(K.LE.2)GO TO 120
        LIST(1,KM1)=NV
        LIST(KL2,KM1)=JC+J
        LIST(KL3,KM1)=IR+J
C END OF J LOOP FROM 60:
120     GO TO 60
C  INCREMENT AND CHECK JC:
125     JC=JC+1
        IF((L1-JC+1).GE.LD)GO TO 40
C  WRITE RESULTS
150     AVE=AVE/COUNT
        DF=COUNT-1.D0
        IF(DF.EQ.0.D0)DF=1.D0
        SIG=(SIG-COUNT*AVE*AVE)/DF
        ALFA=(BIGR+1.)/(COUNT+1.)
        SIG=DSQRT(SIG)
        WRITE(4,1150)COUNT,AVE,SIG
1150    FORMAT(B10.0' DIAGONALS HAVE MEAN ='B9.1'  SIGMA ='B9.1)
        WRITE(4,1151)TS,ALFA
1151    FORMAT(' NON-PARAMETRICS:'/' TARGET SCORE ='F9.0' ALPHA =
     1'1PE13.4)
        WRITE(4,1152)
1152    FORMAT(/'  COORD  DIAG  VALUE   (V-M)/S')
C  LIST LENGTH IS MAXLS:
        MAXLS=100
        DO 160 II=1,MAXLS
        CALL SSW(0,NSW)
        IF(NSW)GO TO 170
        I=MAXLS+1-II
        VN=LIST(1,I)
        SIGN=(VN-AVE)/SIG
        IF(LIST(2,I).EQ.0.AND.LIST(3,I).EQ.0)GO TO 170
        K=LIST(2,I)-LIST(3,I)
        WRITE(4,1160)LIST(2,I),LIST(3,I),K,LIST(1,I),SIGN
1160    FORMAT(2I4,I5,I7,F8.2)
160     CONTINUE
170     GO TO 26
        STOP
        END
 DO 160 II=1,MAXLS
        CALL SSW(0,NSW)
        IF(NSW)GO TO 170
        I=MAXLS+1-II
        VN=LIST(1,I)
        SIGN=(VN-AVE)/SIG
        IF(LIST(2,I).EQ.0.AND.LIST(3,I).EQ.0)GO TO 170
