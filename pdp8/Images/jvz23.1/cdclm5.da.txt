C  CDCLM5.DA
C    CALLS LAGRAN.F4  (LAGRANGE INTERPOLATION)
C     CALLS ROOTCO.F4 (POLYNOMIAL FROM ROOTS)
C         CALCULATES STOKE'S CORRECTION FACTOR
C         FOR MOTION OF TWO SPHERES PARALLEL TO LINE OF CENTERS
C         USING FAXEN'S LAW
C
        PROGRAM STRIP(INPUT,OUTPUT,TAPE5=INPUT,TAPE7=OUTPUT)
        DOUBLE PRECISION ALFA,AINC,AN,Q,Q1,Q2,Q3,X,X1,X2
        DOUBLE PRECISION ALAMDA,DSINH,DCOSH,DTANH,DLOG,DSQRT
        DOUBLE PRECISION S(20),FUDGE,BX(50),BAF(50),BAF2(50),BPOL(50)
        DOUBLE PRECISION C(50),P(50),H(21)
C
        MSAX=20
        MDAT=50
        FUDGE=1.D-25
C
        READ(5,999)N0,N1
999     FORMAT(2I3)
        READ(5,2000)MS
2000    FORMAT(I3)
        IF(MS.GT.MSAX)GOTO 900
        READ(5,2010)(S(I),I=1,MS)
2010    FORMAT(D25.18)
C
        WRITE(7,1010)
        WRITE(7,1011)
C
        DO 700 N=N0,N1
        NDAT=0
        NSTRP=-1
        X=0.D0
        Q=N
        AINC=1.D0/(Q+1.D0)
C
100     X=X+AINC
        IF(X.GT.1.D0-AINC)GOTO 400
C        N=N-1
        X1=1/X
C   X1 = L/2A = COSH(ALPHA)   BRENNER PG. 272
        ALFA=DLOG(X1+DSQRT(X1*X1-1))
        ALAMDA=0.D0
        AN=0.D0
C   START LOOP
200     AN=AN+1.D0
        Q=AN+.5D0
        Q=DSINH(Q*ALFA)
        Q=Q*Q*4.D0
        Q1=2.D0*AN+1.D0
        Q=Q-Q1*Q1*DSINH(ALFA)*DSINH(ALFA)
        Q1=2.D0*DSINH(Q1*ALFA)+Q1*DSINH(2.D0*ALFA)
        Q=1-Q/Q1
        Q1=AN*AN+AN
        Q1=Q1/(4.D0*Q1-3.D0)
        Q=Q1*Q
        ALAMDA=ALAMDA+Q
        IF(Q.GT.FUDGE)GOTO 200
        ALAMDA=ALAMDA*DSINH(ALFA)*4.D0/3.D0
C
C   RESULTS
C
        X2=X/4.D0
C        ALAMDA=1.D0/ALAMDA
C
        NDAT=NDAT+1
        BX(NDAT)=X2
        BAF2(NDAT)=ALAMDA
        GOTO 100
C
C
400     NSTRP=NSTRP+1
        IF(NSTRP.GT.MS)GOTO 650
C
        DO 600 I=1,N
        Q=BAF2(I)
        X=BX(I)
        IF(NSTRP.EQ.0)GOTO 550
        DO 500 J=1,NSTRP
        Q=Q-S(J)*X**(J-1)
500     CONTINUE
        Q=Q/X**(NSTRP)
550     BAF(I)=Q
600     CONTINUE
C
        CALL LAGRAN(MDAT,N,BX,BAF,BPOL,C,P)
        H(NSTRP+1)=BPOL(1)
        GOTO 400
C
650     WRITE(7,1012)N,AINC
        NQ=MS+1
        WRITE(7,1000)(H(I),I=1,NQ)
700     CONTINUE
C
1000    FORMAT(5D25.18)
1010    FORMAT(42H1LAGRANGE INTERPOLATION OF PARALLEL MOTION)
1011    FORMAT(33H OF TWO SPHERES IN VISCIOUS FLUID)
1012    FORMAT(/14H DATA POINTS =,I5,14H   INCREMENT =,D15.8)
C
900     STOP
        END
C
C
C
C  LAGRA2.F4     LAGRANGE INTERPOLATION
C     CALLS ROOTCO.F4 (RETURNS POLYNOMIAL FROM ROOTS)
        SUBROUTINE LAGRAN(MDAT,N,A,F,PSUM,C,P)
        DOUBLE PRECISION A(MDAT),F(MDAT),PSUM(MDAT),C(MDAT),P(MDAT)
C
C    N=# OF DATA POINTS
C    NORD=ORDER OF POLYNOMIAL = N-1
C    A=ORDINATE
C    F=DATA = F(X)
C    POLYNOMIAL INTERPOLATION RETURNED IN PSUM
C
        NORD=N-1
        DO 50 I=1,N
        PSUM(I)=0.D0
50      CONTINUE
C
        DO 500 I=1,N
C
        C(MDAT)=1.D0
        DO 100 J=1,N
        IF(I.EQ.J)GOTO 100
        C(MDAT)=C(MDAT)*(A(I)-A(J))
100     CONTINUE
        C(MDAT)=F(I)/C(MDAT)
C
        J1=0
        DO 200 J=1,N
        IF(I.EQ.J)GOTO 200
        J1=J1+1
        C(J1)=-A(J)
200     CONTINUE
C
        CALL ROOTCO(MDAT,NORD,N,C,P)
C
        DO 300 J=1,N
        PSUM(J)=PSUM(J)+P(J)*C(MDAT)
300     CONTINUE
500     CONTINUE
        RETURN
        END
C
C
C  ROOTCO.F4   COEFFICIENTS OF POLYNOMIAL FROM ROOTS
C      NORD=# OF ROOTS   N=# OF COEF. RETURNED = NORD+1
C
        SUBROUTINE ROOTCO(MDAT,NORD,N,C,P)
        DOUBLE PRECISION C(MDAT),P(MDAT)
C
        P(N)=1.D0
        DO 100 I=1,NORD
        P(I)=0.D0
100     CONTINUE
C
        DO 200 I=1,NORD
        NQ=NORD-I+1
        NQ1=NORD-I+2
        DO 200 J=1,NQ
        NQ2=N-J
        P(NQ2)=P(NQ2)+P(NQ2+1)*C(NQ1-J)
200     CONTINUE
        RETURN
        END
