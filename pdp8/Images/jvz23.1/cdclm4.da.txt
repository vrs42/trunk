C  CDCLM4.DA
C    CALLS TRIDAG.F4  (SOLVES TRIDIAGONAL MATRIX)
C    CALLS LAGRAN.F4  (LAGRANGE INTERPOLATION)
C     CALLS ROOTCO.F4 (POLYNOMIAL FROM ROOTS)
C         CALCULATES STOKE'S CORRECTION FACTOR
C         FOR MOTION OF TWO SPHERES PERPENDICULAR TO LINE OF CENTERS
C         USING METHOD BY GOLDMAN, COX, & BRENNER (NO ROTATION)
C
        PROGRAM STRIP(INPUT,OUTPUT,TAPE5=INPUT,TAPE7=OUTPUT)
        DOUBLE PRECISION ALFA,AINC,Q,X,X1,DI,DI2,RT2,DTRM,BTRM
        DOUBLE PRECISION ALAMDA,DSINH,DCOSH,DTANH,DLOG,DSQRT
        DOUBLE PRECISION A(500,3),B(501),G(501),G0
        DOUBLE PRECISION S(20),FUDGE,BX(50),BAF(50),BAF2(50),BPOL(50)
        DOUBLE PRECISION C(50),P(50),H(21)
C
        MAX=500
        MSAX=20
        MDAT=50
        FUDGE=1.D-25
        RT2=DSQRT(2.D0)
C
        READ(5,2000)MS
2000    FORMAT(I3)
        IF(MS.GT.MSAX)GOTO 900
        READ(5,2010)(S(I),I=1,MS)
2010    FORMAT(D25.18)
C
        WRITE(7,1010)
        WRITE(7,1011)
C
        DO 700 N=5,15
        NDAT=0
        NSTRP=-1
        X=0.D0
        Q=N
        AINC=1.D0/(Q+1.D0)
C
100     X=X+AINC
        IF(X.GT.1.D0-AINC)GOTO 400
        X1=1.D0/X
C   X1 = L/2A = COSH(ALPHA)   BRENNER PG. 272
        ALFA=DLOG(X1+DSQRT(X1*X1-1))
C
        DO 10 I=1,MAX
        B(I)=0.D0
        G(I)=0.D0
        DO 10 J=1,3
10      A(I,J)=0.D0
        B(MAX+1)=0.D0
        G(MAX+1)=0.D0
C
C   START LOOP
C
        G0=DTANH(.5D0*ALFA)/DTANH(ALFA)
        NQ=MAX+1
        DO 200 I=1,NQ
200     G(I)=DTANH((I+.5)*ALFA)/DTANH(ALFA)
C
C   REM CALC. FIRST SET OF COEF. FOR A
        A(1,2)=3.D0-5.D0*G(1)+(10.D0*G(2)-G0-11.D0)/3.D0
        A(1,3)=21.D0*(G(1)+1.D0)/5.D0-3.D0*(G(2)+1.D0)
C
        Q=DEXP(ALFA)/DCOSH(.5D0*ALFA)
        Q=Q-2.D0/DCOSH(1.5D0*ALFA)
        Q=Q+DEXP(-ALFA)/DCOSH(2.5D0*ALFA)
        B(1)=RT2*DEXP(-1.5D0*ALFA)*Q
C
C   CALC. REST OF COEF.
        I=1
250     I=I+1
        IF(I.GT.MAX)GOTO 260
        DI=I
        DI2=2.D0*DI
C
        Q=(DI-1.D0)*(G(I-1)-1.D0)
        A(I,1)=Q-(DI-1.D0)*(DI2-3.D0)*(G(I)-1.D0)/(DI2-1.D0)
C
        Q=DI2+1.D0-5.D0*G(I)
        Q=Q-DI*(DI2-1.D0)*(G(I-1)+1.D0)/(DI2+1.D0)
        A(I,2)=Q+(DI+1.D0)*(DI2+3.D0)*(G(I+1)-1.D0)/(DI2+1.D0)
C
        Q=(DI+2.D0)*(DI2+5.D0)*(G(I)+1.D0)/(DI2+3.D0)
        A(I,3)=Q-(DI+2.D0)*(G(I+1)+1.D0)
C
        Q=DEXP(ALFA)/DCOSH((DI-.5D0)*ALFA)
        Q=Q-2.D0/DCOSH((DI+.5D0)*ALFA)
        Q=Q+DEXP((-1.D0)*ALFA)/DCOSH((DI+1.5D0)*ALFA)
        B(I)=RT2*DEXP((-1.D0)*(DI+.5D0)*ALFA)*Q
C
        IF(B(I).GT.FUDGE)GOTO 250
260     NN=I-1
C
C    SOLVE SET OF SIMULTANEOUS EQS. A()   (RETURNED IN B)
        NQ=MAX+1
        CALL TRIDAG(MAX,NQ,NN,A,B)
C
C    ADD SERIES
C     CALCULATE ZERO TERM
        Q=2.D0*RT2*DEXP((-.5D0)*ALFA)/DCOSH(.5D0*ALFA)
        ALAMDA=Q+2.D0*(G0+1.D0)*B(1)/3.D0
C     CALC. REST OF TERMS
        DO 300 I=1,NN
        DI=I
        DI2=2.D0*DI
        II=I-1
        IF(II.LE.0)II=NN+1
C
        DTRM=2.D0*RT2*DEXP((-1.D0)*(DI+.5D0)*ALFA)
        DTRM=DTRM/DCOSH((DI+.5D0)*ALFA)
        DTRM=DTRM-DI*(DI-1.D0)*(G(I)-1.D0)*B(II)/(DI2-1.D0)
        DTRM=DTRM+(DI+1.D0)*(DI+2.D0)*(G(I)+1.D0)*B(I+1)/(DI2+3.D0)
C
        BTRM=2.D0*(DI-1.D0)*(G(I)-1.D0)*B(II)/(DI2-1.D0)
        BTRM=BTRM-2.D0*G(I)*B(I)
        BTRM=BTRM+2.D0*(DI+2.D0)*(G(I)+1.D0)*B(I+1)/(DI2+3.D0)
C
        ALAMDA=ALAMDA+DTRM+DI*(DI+1.D0)*BTRM
300     CONTINUE
        ALAMDA=ALAMDA*DSINH(ALFA)*(-1.D0)*RT2/6.D0
C
C   RESULTS
C
        X2=X/2.D0
        ALAMDA=1./ALAMDA
        ALAMDA=-ALAMDA
C
        NDAT=NDAT+1
        BX(NDAT)=X2
        BAF2(NDAT)=ALAMDA
        GOTO 100
C
C
400     NSTRP=NSTRP+1
        IF(NSTRP.GT.MS)GOTO 650
C
        DO 600 I=1,N
        Q=BAF2(I)
        X=BX(I)
        IF(NSTRP.EQ.0)GOTO 550
        DO 500 J=1,NSTRP
        Q=Q-S(J)*X**(J-1)
500     CONTINUE
        Q=Q/X**(NSTRP)
550     BAF(I)=Q
600     CONTINUE
C
        CALL LAGRAN(MDAT,N,BX,BAF,BPOL,C,P)
        H(NSTRP+1)=BPOL(1)
        GOTO 400
C
650     WRITE(7,1012)N,AINC
        NQ=MS+1
        WRITE(7,1000)(H(I),I=1,NQ)
700     CONTINUE
C
1000    FORMAT(5D25.18)
1010    FORMAT(47H1LAGRANGE INTERPOLATION OF PERPENDICULAR MOTION)
1011    FORMAT(33H OF TWO SPHERES IN VISCIOUS FLUID)
1012    FORMAT(/14H DATA POINTS =,I5,14H   INCREMENT =,D15.8)
C
900     STOP
        END
C   TRIDAG.F4
C     SOLVES A SET OF SIMULTANEOUS EQUATIONS
C     IN TRI-DIAGONAL FORM
C     USING GAUSSIAN REDUCTION
C      FIRST AND LAST ELEMENTS OF MATRIX MUST BE ZERO
C
        SUBROUTINE TRIDAG(NMAX,N1,N,A,B)
        DOUBLE PRECISION A(NMAX,3),B(N1)
C
        A(1,1)=0.D0
        A(N,3)=0.D0
C
        B(1)=B(1)/A(1,2)
        A(1,3)=A(1,3)/A(1,2)
        A(1,2)=1.D0
C
        DO 100 I=2,N
        A(I,2)=A(I,2)-A(I,1)*A(I-1,3)
        B(I)=B(I)-A(I,1)*B(I-1)
        A(I,1)=0.D0
C
        A(I,3)=A(I,3)/A(I,2)
        B(I)=B(I)/A(I,2)
        A(I,2)=1.D0
100     CONTINUE
C
        NQ=N-1
        DO 200 I=1,NQ
        II=N-I
        B(II)=B(II)-A(II,3)*B(II+1)
200     CONTINUE
        RETURN
        END
C  LAGRA2.F4     LAGRANGE INTERPOLATION
C     CALLS ROOTCO.F4 (RETURNS POLYNOMIAL FROM ROOTS)
        SUBROUTINE LAGRAN(MDAT,N,A,F,PSUM,C,P)
        DOUBLE PRECISION A(MDAT),F(MDAT),PSUM(MDAT),C(MDAT),P(MDAT)
C
C    N=# OF DATA POINTS
C    NORD=ORDER OF POLYNOMIAL = N-1
C    A=ORDINATE
C    F=DATA = F(X)
C    POLYNOMIAL INTERPOLATION RETURNED IN PSUM
C
        NORD=N-1
        DO 50 I=1,N
        PSUM(I)=0.D0
50      CONTINUE
C
        DO 500 I=1,N
C
        C(MDAT)=1.D0
        DO 100 J=1,N
        IF(I.EQ.J)GOTO 100
        C(MDAT)=C(MDAT)*(A(I)-A(J))
100     CONTINUE
        C(MDAT)=F(I)/C(MDAT)
C
        J1=0
        DO 200 J=1,N
        IF(I.EQ.J)GOTO 200
        J1=J1+1
        C(J1)=-A(J)
200     CONTINUE
C
        CALL ROOTCO(MDAT,NORD,N,C,P)
C
        DO 300 J=1,N
        PSUM(J)=PSUM(J)+P(J)*C(MDAT)
300     CONTINUE
500     CONTINUE
        RETURN
        END
C  ROOTCO.F4   COEFFICIENTS OF POLYNOMIAL FROM ROOTS
C      NORD=# OF ROOTS   N=# OF COEF. RETURNED = NORD+1
C
        SUBROUTINE ROOTCO(MDAT,NORD,N,C,P)
        DOUBLE PRECISION C(MDAT),P(MDAT)
C
        P(N)=1.D0
        DO 100 I=1,NORD
        P(I)=0.D0
100     CONTINUE
C
        DO 200 I=1,NORD
        NQ=NORD-I+1
        NQ1=NORD-I+2
        DO 200 J=1,NQ
        NQ2=N-J
        P(NQ2)=P(NQ2)+P(NQ2+1)*C(NQ1-J)
200     CONTINUE
        RETURN
        END
 = NORD+1
C
        SUBROUTINE ROOTCO(MDAT,NORD,N,C,P)
        DOUBLE PRECISION C(MDAT),P(MDAT)
C
        P(N)=1.D0
        DO 100 I=1,NORD
        P(I)=0.D0
100     CONTINUE
C
        DO 200 I=1,NORD
        NQ=NORD-I+1
        NQ1=NORD-I+2
        DO 200 J=1,NQ
        NQ2=N-J
        P(NQ2)=P(NQ2)+P(NQ2+1)*C(NQ1-J)
200     CONTINUE
